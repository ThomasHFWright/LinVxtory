+unsafe fn copy_color(
+    list: &HostList,
+    texture: &ID3D12Resource,
+    readback: &ID3D12Resource,
+    offset: u64,
+) {
+    // SAFETY: same device, native target and matching single-subresource readback footprint.
+    unsafe {
+        let from = D3D12_TEXTURE_COPY_LOCATION {
+            pResource: std::mem::transmute_copy(texture),
+            Type: D3D12_TEXTURE_COPY_TYPE_SUBRESOURCE_INDEX,
+            Anonymous: D3D12_TEXTURE_COPY_LOCATION_0 {
+                SubresourceIndex: 0,
+            },
+        };
+        let to = D3D12_TEXTURE_COPY_LOCATION {
+            pResource: std::mem::transmute_copy(readback),
+            Type: D3D12_TEXTURE_COPY_TYPE_PLACED_FOOTPRINT,
+            Anonymous: D3D12_TEXTURE_COPY_LOCATION_0 {
+                PlacedFootprint: D3D12_PLACED_SUBRESOURCE_FOOTPRINT {
+                    Offset: offset,
+                    Footprint: D3D12_SUBRESOURCE_FOOTPRINT {
+                        Format: HDR,
+                        Width: WIDTH,
+                        Height: HEIGHT,
+                        Depth: 1,
+                        RowPitch: PITCH,
+                    },
+                },
+            },
+        };
+        let target = D3D12_RESOURCE_STATE_RENDER_TARGET;
+        let source = D3D12_RESOURCE_STATE_COPY_SOURCE;
+        list.list.barriers(&[transition(texture, target, source)]);
+        list.list.CopyTextureRegion(&to, 0, 0, 0, &from, None);
+        list.list.barriers(&[transition(texture, source, target)]);
+    }

+pub(crate) fn submitted(
+    runs: &[&HostList],
+    submission: u64,
+    background: Option<&background_snapshot::RrBackgroundSnapshot>,
+) -> Option<Vec<Images>> {
+    if frame().is_none() {
+        return None;
+    }
+    let interval = background.and_then(|background| {
+        background.camera.as_ref()?;
+        let [begin, end] = background.producer_epochs();
+        let left = runs.iter().position(|list| list.epoch() == begin)?;
+        let right = runs.iter().rposition(|list| list.epoch() == end)?;
+        (right > left + 1).then_some(((left + 1)..right, background))
+    });
+    for (i, list) in runs.iter().enumerate() {
+        let matching = interval
+            .as_ref()
+            .and_then(|(range, background)| range.contains(&i).then_some(*background));
+        list.mark_transmission(submission, matching);
+    }
+    if !overlay_test() {
+        return None;
+    }
+    let (range, _) = interval?;
+    let mut images = Vec::new();
+    for list in &runs[range] {
+        list.transmission_overlay_images(&mut images)?;
+        if images.len() > MAX_DRAWS_PER_LIST {
+            return None;
+        }
+    }
+    (!images.is_empty()).then_some(images)
+}

+fn changed_color(before: &[u16], after: &[u16]) -> Footprint {
+    let mut footprint = Footprint {
+        mask: vec![0; before.len() / 4],
+        changed: 0,
+        nonfinite: 0,
+        bounds: None,
+    };
+    let canonical_zero = |value: u16| if value & 0x7fff == 0 { 0 } else { value };
+    for (pixel, (before, after)) in before
+        .chunks_exact(4)
+        .zip(after.chunks_exact(4))
+        .enumerate()
+    {
+        if before[..3]
+            .iter()
+            .chain(&after[..3])
+            .any(|&word| word & 0x7c00 == 0x7c00)
+        {
+            footprint.nonfinite += 1;
+            continue;
+        }
+        if before[..3]
+            .iter()
+            .zip(&after[..3])
+            .all(|(&a, &b)| canonical_zero(a) == canonical_zero(b))
+        {
+            continue;
+        }
+        let x = pixel as u32 % WIDTH;
+        let y = pixel as u32 / WIDTH;
+        let [left, top, right, bottom] = footprint.bounds.unwrap_or([x, y, x + 1, y + 1]);
+        footprint.bounds = Some([left.min(x), top.min(y), right.max(x + 1), bottom.max(y + 1)]);
+        footprint.mask[pixel] = 255;
+        footprint.changed += 1;
+    }
+    footprint
+}

+///
+/// # Safety
+/// Both HDR snapshots come from the same frame, camera, native geometry and

+    list: &HostList,
+    final_hdr: &ID3D12Resource,
+    without_transmission: &ID3D12Resource,
+    coverage: &ID3D12Resource,
