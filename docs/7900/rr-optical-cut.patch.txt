+fn serial(title: usize, format: i32, size: [u32; 2]) -> Option<u64> {

+        entry.renewed = entry.renewed.max(sequence);
+        true
+    }
+}

+const BYTES: usize = size_of::<D3D12_QUERY_DATA_PIPELINE_STATISTICS>();
+
+pub(super) struct Query {
+    heap: ID3D12QueryHeap,
+    readback: ID3D12Resource,
+    frame: u64,
+}
+
+impl Query {
+    pub(super) fn new(list: &HostList) -> Option<Self> {
+        let host = crate::host::host()?;
+        let mut heap = None;
+        // SAFETY: the query slots belong to this pair and are used by the one recording.
+        unsafe {
+            host.device
+                .CreateQueryHeap(
+                    &D3D12_QUERY_HEAP_DESC {
+                        Type: D3D12_QUERY_HEAP_TYPE_PIPELINE_STATISTICS,
+                        Count: 2,
+                        NodeMask: 0,
+                    },
+                    &mut heap,
+                )
+                .ok()?;
+        }
+        let heap: ID3D12QueryHeap = heap?;
+        let readback = host.read_back_buffer((BYTES * 2) as u64)?;
+        // Verify the pair produced both query results, following pass_stats' private readback sentinel.
+        let mut data = std::ptr::null_mut();
+        unsafe {
+            readback.Map(0, None, Some(&mut data)).ok()?;
+            if data.is_null() {
+                return None;
+            }
+            std::ptr::write_bytes(data.cast::<u8>(), 0xff, BYTES * 2);
+            readback.Unmap(0, None);
+        }
+        list.keep(&heap);
+        list.keep(&readback);
+        Some(Self {
+            heap,
+            readback,
+            frame: crate::queue::presented() + 1,
+        })
+    }
+
+    pub(super) fn begin(&self, list: &HostList, slot: u32) {
+        // SAFETY: the slot is private to the open list's paired filter.
+        unsafe {
+            list.list
+                .BeginQuery(&self.heap, D3D12_QUERY_TYPE_PIPELINE_STATISTICS, slot)
+        };
+    }
+
+    pub(super) fn end(&self, list: &HostList, slot: u32) {
+        // SAFETY: closes the preceding query in the same list, into a disjoint readback slot.
+        unsafe {
+            list.list
+                .EndQuery(&self.heap, D3D12_QUERY_TYPE_PIPELINE_STATISTICS, slot);
+            list.list.ResolveQueryData(
+                &self.heap,
+                D3D12_QUERY_TYPE_PIPELINE_STATISTICS,
+                slot,
+                1,
+                &self.readback,
+                BYTES as u64 * u64::from(slot),
+            );
+        }
+    }
+
+    fn read(&self) -> Option<[u64; 2]> {
+        let mut data = std::ptr::null_mut();
+        // SAFETY: the caller proved native fence completion for both queries.
+        unsafe {
+            let range = D3D12_RANGE {
+                Begin: 0,
+                End: BYTES * 2,
+            };
+            self.readback.Map(0, Some(&range), Some(&mut data)).ok()?;
+            if data.is_null() {
+                return None;
+            }
+            let values: [u64; 2] = std::array::from_fn(|i| {
+                let ptr = data
+                    .cast::<u8>()
+                    .add(i * BYTES)
+                    .cast::<D3D12_QUERY_DATA_PIPELINE_STATISTICS>();
+                (*ptr).CSInvocations
+            });
+            self.readback
+                .Unmap(0, Some(&D3D12_RANGE { Begin: 0, End: 0 }));
+            Some(values)
+        }
+    }
+}

+// vkd3d-proton's indirect argument masking can contribute 32 invocations
+// without running the title shader. The calibrated native kernels exceed it.
+fn executed(values: [u64; 2]) -> bool {
+    let [bilateral, history] = values;
+    bilateral == history && history > 32 && history != u64::MAX
+}

+pub(super) struct Accepted {
+    pub fence: ID3D12Fence,
+    pub value: u64,
+    pub sequence: u64,
+    pub frame: u64,
+    pub list_index: usize,
+    pub list_epoch: u64,

+pub(super) fn retire(query: &Query, pair: CutPair, cut: Option<u64>, sent: Accepted) {
+    // SAFETY: the accepted host submission retained its queue fence.
+    let completed = unsafe { sent.fence.GetCompletedValue() };
+    if completed == u64::MAX || completed < sent.value {
+        return;
+    }
+    let Some(invocations) = query.read() else {
+        return;
+    };
+    if crate::host::pass_stats::selected(query.frame) {
+        for ((slot, shader), cs_invocations) in [
+            (0, super::opaque_current::BILATERAL),
+            (1, super::opaque_current::HISTORY),
+        ]
+        .into_iter()
+        .zip(invocations)
+        {
+            if cs_invocations != u64::MAX {
+                tracing::info!(
+                    frame = sent.frame,
+                    submission = sent.sequence,
+                    list_index = sent.list_index,
+                    list_epoch = sent.list_epoch,
+                    slot,
+                    recorded_frame = query.frame,
+                    shader,
+                    max_commands = 1,
+                    fence = sent.value,
+                    cs_invocations,
+                    "retired pass statistics"
+                );
+            }
+        }
+    }
+    if let Some(cut) = cut
+        && executed(invocations)
+        && lock().renew(pair, cut, sent.sequence)
+    {
+        tracing::info!(
+            sequence = sent.sequence,
+            fence = sent.value,
+            cut,
+            packed = pair.packed,
+            alias = pair.packed_alias,
+            bilateral = invocations[0],
+            history = invocations[1],
+            "RR opaque native history update fence-retired"
+        );
+    }
+}

+pub(super) fn drawn(
+    state: &State,
+    list: &HostList,
+    pipeline: Option<&HostPipeline>,
+    full_triangle: bool,
+    args: Option<u64>,
+) {
+    if !enabled() {
+        return;
+    }
+    let reject = |outcome, serial| {
+        if witness_enabled()
+            && let Some(p) = pipeline
+            && let [vertex, pixel] = p.stages.as_slice()
+        {
+            let stage = match [vertex.shader.hash.as_str(), pixel.shader.hash.as_str()] {
+                COMPOSITE => Some("tone"),
+                DITHER => Some("dither"),
+                HAIR_BLUR => Some("hair_blur"),
+                HAIR_COPY => Some("hair_copy"),
+                _ => None,
+            };
+            if let Some(stage) = stage { witness(stage, outcome, serial, 0); }
+        }
+        None
+    };

+        if state.target_handles.len() != 1 {
+            return reject("unexpected_target_count", 0);
+        }
+        let [title] = state.targets.as_slice() else {
+            return reject("untracked_target", 0);
+        };
+        let Some(dest) = target(*title as This) else {
+            return reject("untracked_target", 0);
+        };
+        let p = pipeline?;
+        let [vertex, pixel] = p.stages.as_slice() else {
+            return None;
+        };
+        let signatures = [vertex.shader.hash.as_str(), pixel.shader.hash.as_str()];
+        let [vp] = state.viewports.as_slice() else {
+            return reject("unexpected_raster_state", dest.serial);
+        };
+        let [scissor] = state.scissors.as_slice() else {
+            return reject("unexpected_raster_state", dest.serial);
+        };
+        if state.depth != 0
+            || !p.state.contains("depth false")
+            || !p.state.contains("stencil false")
+            || !p.state.contains("blend false")
+            || !(p.state.contains("mask 0x7") || p.state.contains("mask 0xf"))
+            || [vp.TopLeftX, vp.TopLeftY, vp.Width, vp.Height] != [0., 0., 3840., 2160.]
+            || [scissor.left, scissor.top, scissor.right, scissor.bottom] != [0, 0, 3840, 2160]
+        {
+            return reject("unexpected_raster_state", dest.serial);
+        }
+        let (pass, slot, offset, from, to) = match (signatures, full_triangle, args) {
+            (COMPOSITE, true, _) => (Pass::Tone, 1, 304, HDR, TONE),
+            (DITHER, true, _) => (Pass::Dither, 1, 0, TONE, BASE),
+            (HAIR_BLUR, false, Some(args)) => (Pass::HairBlur { args }, 0, 40, HDR, HDR),
+            (HAIR_COPY, false, Some(args)) => (Pass::HairCopy { args }, 0, 16, HDR, HDR),
+            _ => return reject("unexpected_draw_arguments", dest.serial),
+        };
+        let Some(src) = source(state, list, slot, offset) else {
+            witness(pass.report(), "unbound_source", dest.serial, 0);
+            return None;
+        };
+        if !(format(&src, from) && format(&dest, to) && src.serial != dest.serial) {
+            witness(pass.report(), "unexpected_view_format", dest.serial, 0);
+            return None;
+        }
+        Some(Event::Transfer(pass, src, dest))
+    };
+    if let Some(event) = transfer() {
+        list.present_lineage_event(event);
+    } else {
+        for &title in &state.targets {
+            overwritten(list, title as This);
+        }
+    }
+}

+struct Tagged {
+    backing: Arc<HostTexture>,
+    stage: Stage,
+    receipt: ProviderReceipt,
+}

+static SIGNED: Mutex<Option<HashMap<usize, Tagged>>> = Mutex::new(None);

+fn read(map: &HashMap<usize, Tagged>, copy: &HostTexture) -> Option<(Stage, ProviderReceipt)> {
+    map.get(&identity(copy))
+        .filter(|tagged| tagged.backing.serial == copy.serial)
+        .map(|tagged| (tagged.stage, tagged.receipt))
+}

+fn transfer(
+    map: &mut HashMap<usize, Tagged>,
+    pass: Pass,
+    src: Arc<HostTexture>,
+    dest: Arc<HostTexture>,
+    submit: u64,
+) {
+    let previous = map.remove(&identity(&dest));
+    let Some((stage, mut receipt)) = read(map, &src) else {
+        witness(pass.report(), "missing_source_tag", src.serial, submit);
+        return;
+    };
+    let output = match (pass, stage) {
+        (Pass::Tone, Stage::Hdr) => {
+            map.remove(&identity(&src));
+            receipt.compositor_submission = submit;
+            Stage::Tone
+        }
+        (Pass::Dither, Stage::Tone) => {
+            map.remove(&identity(&src));
+            receipt.final_submission = submit;
+            Stage::Base
+        }
+        (Pass::HairBlur { args }, Stage::Hdr) => Stage::Hair {
+            origin: src.serial,
+            args,
+        },
+        (
+            Pass::HairCopy { args },
+            Stage::Hair {
+                origin,
+                args: source_args,
+            },
+        ) if origin == dest.serial
+            && source_args == args
+            && previous.is_some_and(|tag| tag.stage == Stage::Hdr && tag.receipt == receipt) =>
+        {
+            map.remove(&identity(&src));
+            Stage::Hdr
+        }
+        _ => {
+            witness(pass.report(), "incompatible_source_tag", src.serial, submit);
+            return;
+        }
+    };
+    witness(pass.report(), "linked", dest.serial, submit);
+    map.insert(
+        identity(&dest),
+        Tagged {
+            backing: dest,
+            stage: output,
+            receipt,
+        },
+    );
+}

+pub(crate) fn submitted(
+    events: Vec<Event>,
+    overflow: bool,
+    rr_output: Option<(u64, [bool; 4])>,
+    submit: u64,
+    fence: u64,
+) {
+    if events.is_empty() && !overflow {
+        return;
+    }
+    let mut signed = SIGNED.lock().unwrap_or_else(|e| e.into_inner());
+    let map = signed.get_or_insert_with(HashMap::new);
+    if overflow || map.len() > 64 {
+        witness("submission", "invalid_or_overflow", 0, submit);
+        map.clear();
+        return;
+    }
+    map.retain(|_, tag| submit.saturating_sub(tag.receipt.ffx_submission) <= 128);
+    for event in events {
+        match event {
+            Event::Seed(copy, begun) => {
+                if rr_output.is_some_and(|(serial, _)| serial != copy.serial) {
+                    witness("ffx", "replacement_backing_mismatch", copy.serial, submit);
+                    map.remove(&identity(&copy));
+                    continue;
+                }
+                let [early_cut, optical, paired_alpha, paired_probe] =
+                    rr_output.map_or([false; 4], |(_, mode)| mode);
+                let receipt = ProviderReceipt {
+                    is_rr: rr_output.is_some(),
+                    early_cut,
+                    optical,
+                    paired_alpha,
+                    paired_probe,
+                    begun,
+                    ffx_submission: submit,
+                    ffx_fence: fence,
+                    hdr_serial: copy.serial,
+                    compositor_submission: 0,
+                    final_submission: 0,
+                    present_submission: 0,
+                    present_fence: 0,
+                };
+                witness("ffx", "linked", copy.serial, submit);
+                map.insert(
+                    identity(&copy),
+                    Tagged {
+                        backing: copy,
+                        stage: Stage::Hdr,
+                        receipt,
+                    },
+                );
+            }
+            Event::UnknownWrite(copy) => {
+                if map.remove(&identity(&copy)).is_some() {
+                    witness("submission", "tag_overwritten", copy.serial, submit);
+                }
+            }
+            Event::Transfer(pass, src, dest) => transfer(map, pass, src, dest, submit),
+            Event::Present(key, probe) => {
+                let result = match map.remove(&key) {
+                    Some(mut tag) if tag.stage == Stage::Base && tag.receipt.final_submission < submit => {
+                        tag.receipt.present_submission = submit;
+                        tag.receipt.present_fence = fence;
+                        Some((tag.receipt, tag.backing.serial))
+                    }
+                    Some(tag) => {
+                        witness("present", "incorrect_native_base", tag.backing.serial, submit);
+                        None
+                    }
+                    None => {
+                        witness("present", "missing_fresh_native_base", 0, submit);
+                        None
+                    }
+                };
+                *probe.lock().unwrap_or_else(|e| e.into_inner()) = result;
+            }
+        }
+    }
+    // This call covers exactly one closed physical list. The blur and copy
+    // may change binding-cache epochs, but the intermediate cannot cross lists.
+    map.retain(|_, tag| !matches!(tag.stage, Stage::Hair { .. }));
+}

+pub(super) fn defer(list: &HostList, dest: *mut u8, source: u64, request: Arc<Request>) {

+                tracing::info!(?request.key, "screen probe current-only kernel selected");
+            }
+        });
+    }

+        use crate::host::pass_stats::{self, Variant};

+                selected: self.current.clone(),

+    }
