+//! Bounded CPU-side texture binding history, written in host submission order.
+//! shortcut: accesses cover texture bindings and title copies at resource
+//! granularity, add GPU-resolved descriptor indices and subresource coverage
+//! when candidate bindings cannot establish the required chain.
+
+use std::collections::HashMap;

+const MAX_PASSES_PER_LIST: usize = 2048;
+const MAX_BINDINGS: usize = 128;
+const MAX_ROWS_PER_FRAME: usize = 200_000;
+
+struct Capture {
+    warmup: u64,
+    last: u64,
+    path: std::ffi::OsString,
+}
+
+fn capture() -> Option<&'static Capture> {
+    static CAPTURE: OnceLock<Option<Capture>> = OnceLock::new();
+    CAPTURE
+        .get_or_init(|| {
+            let path = std::env::var_os("VICTORY_PROVENANCE")?;
+            let frames = std::env::var("VICTORY_PROVENANCE_FRAMES").ok()?;
+            let (first, last) = frames.split_once('-').unwrap_or((&frames, &frames));
+            let (first, last): (u64, u64) = (first.parse().ok()?, last.parse().ok()?);
+            (last.checked_sub(first)? < 8).then_some(Capture {
+                warmup: first.saturating_sub(2),
+                last,
+                path,
+            })
+        })
+        .as_ref()
+}
+
+fn frame() -> u64 {
+    crate::queue::presented() + 1
+}
+
+pub(crate) fn enabled() -> bool {
+    capture().is_some()
+}
+
+pub(crate) fn active() -> bool {
+    capture().is_some_and(|c| (c.warmup..=c.last).contains(&frame()))
+}

+#[derive(Default)]
+pub(crate) struct Recording {
+    passes: Vec<Pass>,
+    truncated: bool,
+}

+pub(crate) struct Pass {
+    name: &'static str,
+    recorded: u64,
+    pipeline: usize,
+    shaders: String,
+    bindings: Vec<Binding>,
+    truncated: bool,
+}

+struct Binding {
+    role: &'static str,
+    read: bool,
+    write: bool,
+    slot: u32,
+    offset: u32,
+    title: usize,
+    serial: u64,
+    host: usize,
+    dimensions: [u64; 3],
+    format: i32,
+    view: Option<u32>,
+}

+impl Pass {
+    pub(crate) fn new(name: &'static str, pipeline: usize) -> Self {
+        Self {
+            name,
+            recorded: frame(),
+            pipeline,
+            shaders: String::new(),
+            bindings: Vec::new(),
+            truncated: false,
+        }
+    }
+
+    pub(crate) fn shader(&mut self, hash: &str) {
+        if !self.shaders.is_empty() {
+            self.shaders.push(',');
+        }
+        self.shaders.push_str(hash);
+    }
+
+    pub(crate) fn texture(
+        &mut self,
+        role: &'static str,
+        (read, write): (bool, bool),
+        (slot, offset): (u32, u32),
+        title: This,
+        view: Option<u32>,
+    ) {
+        let Some(copy) = crate::resource::made_copy(title) else {
+            return;
+        };
+        if self.bindings.len() == MAX_BINDINGS {
+            self.truncated = true;
+            return;
+        }
+        // SAFETY: the Arc keeps the backing allocation alive.
+        let d = unsafe { copy.texture.GetDesc() };
+        self.bindings.push(Binding {
+            role,
+            read,
+            write,
+            slot,
+            offset,
+            title: title as usize,
+            serial: copy.serial,
+            host: copy.texture.as_raw() as usize,
+            dimensions: [d.Width, d.Height.into(), d.DepthOrArraySize.into()],
+            format: d.Format.0,
+            view,
+        });
+    }
+}

+impl Recording {
+    pub(crate) fn push(&mut self, pass: Pass) {
+        if pass.bindings.is_empty() {
+            return;
+        }
+        if self.passes.len() == MAX_PASSES_PER_LIST {
+            self.truncated = true;
+            return;
+        }
+        self.passes.push(pass);
+    }

+pub(crate) fn transfer(list: &HostList, name: &'static str, source: This, target: This) {
+    if active() {
+        let mut pass = Pass::new(name, 0);
+        pass.texture("source", (true, false), (0, 0), source, None);
+        pass.texture("target", (false, true), (0, 0), target, None);
+        list.provenance(pass);
+    }

+type Versions = HashMap<u64, u64>;
+
+fn advance(versions: &mut Versions, accesses: impl Iterator<Item = (u64, bool)>) {
+    // One new version per written resource, even if several bindings alias it.
+    let mut written = [0; MAX_BINDINGS];
+    let mut count = 0;
+    for (serial, write) in accesses {
+        if write && !written[..count].contains(&serial) {
+            written[count] = serial;
+            count += 1;
+            *versions.entry(serial).or_default() += 1;
+        }
+    }
+}

+struct Output {
+    file: BufWriter<std::fs::File>,
+    versions: Versions,
+    rows: usize,
+    frame: u64,
+    failed: bool,

+pub(crate) fn submitted(sequence: u64, lists: &[&HostList]) {
+    if !active() {
+        return;
+    }
+    static OUTPUT: OnceLock<Option<Mutex<Output>>> = OnceLock::new();
+    let Some(output) = OUTPUT.get_or_init(|| {
+        let file = match std::fs::File::create(&capture()?.path) {
+            Ok(file) => file,
+            Err(error) => {
+                tracing::warn!(%error, "could not open texture provenance output");
+                return None;
+            }
+        };
+        let mut file = BufWriter::new(file);
+        writeln!(file, "frame\tsubmit\tlist_index\tlist_epoch\tpass\trecorded_frame\tkind\tpipeline\tshaders\trole\tslot\toffset\ttitle\tserial\thost\twidth\theight\tdepth_or_array\tresource_format\tview_format\treads\twrites").ok()?;
+        Some(Mutex::new(Output { file, versions: Versions::new(), rows: 0, frame: 0, failed: false }))
+    }) else {
+        return;
+    };
+    let mut output = output.lock().unwrap_or_else(|e| e.into_inner());
+    let frame = frame();
+    if output.frame != frame {
+        output.frame = frame;
+        output.rows = 0;
+    }
+    if output.failed || output.rows >= MAX_ROWS_PER_FRAME {
+        return;
+    }
+    if let Err(error) = output.write_submission(frame, sequence, lists) {
+        output.failed = true;
+        tracing::warn!(%error, "could not write texture provenance output");
+    }
+}

+impl Output {
+    fn write_submission(
+        &mut self,
+        frame: u64,
+        sequence: u64,
+        lists: &[&HostList],
+    ) -> std::io::Result<()> {
+        for (list_index, list) in lists.iter().enumerate() {
+            let recording = list.provenance_recording();
+            let epoch = list.epoch();
+            if recording.truncated {
+                writeln!(
+                    self.file,
+                    "# truncated list {sequence}/{list_index}/{epoch}"
+                )?;
+            }
+            for (order, pass) in recording.passes.iter().enumerate() {
+                if self.rows + pass.bindings.len() >= MAX_ROWS_PER_FRAME {
+                    writeln!(
+                        self.file,
+                        "# truncated frame {frame} at {MAX_ROWS_PER_FRAME} rows"
+                    )?;
+                    self.rows = MAX_ROWS_PER_FRAME;
+                    return self.file.flush();
+                }
+                if pass.truncated {
+                    writeln!(
+                        self.file,
+                        "# truncated bindings {sequence}/{list_index}/{order}"
+                    )?;
+                }
+                let mut prefix = String::new();
+                let _ = write!(
+                    prefix,
+                    "{frame}\t{sequence}\t{list_index}\t{epoch}\t{order}\t{}\t{}\t{:#x}\t{}",
+                    pass.recorded, pass.name, pass.pipeline, pass.shaders
+                );
+                for b in &pass.bindings {
+                    let old = self.versions.get(&b.serial).copied().unwrap_or(0);
+                    let [w, h, d] = b.dimensions;
+                    writeln!(
+                        self.file,
+                        "{prefix}\t{}\t{}\t{}\t{:#x}\t{}\t{:#x}\t{w}\t{h}\t{d}\t{}\t{}\t{}\t{}",
+                        b.role,
+                        b.slot,
+                        b.offset,
+                        b.title,
+                        b.serial,
+                        b.host,
+                        b.format,
+                        b.view.map(|v| v.to_string()).unwrap_or_default(),
+                        b.read.then(|| old.to_string()).unwrap_or_default(),
+                        b.write.then(|| (old + 1).to_string()).unwrap_or_default(),
+                    )?;
+                }
+                self.rows += pass.bindings.len();
+                advance(
+                    &mut self.versions,
+                    pass.bindings.iter().map(|b| (b.serial, b.write)),
+                );
+            }
+        }
+        self.file.flush()
+    }
+}
