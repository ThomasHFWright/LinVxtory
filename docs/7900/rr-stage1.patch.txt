+fn is_color(desc: D3D12_RESOURCE_DESC) -> bool {
+    desc.Dimension == D3D12_RESOURCE_DIMENSION_TEXTURE2D
+        && desc.Width == u64::from(WIDTH)
+        && desc.Height == HEIGHT
+        && desc.DepthOrArraySize == 1
+        && desc.MipLevels == 1
+        && desc.SampleDesc.Count == 1
+        && desc.Format == DXGI_FORMAT_R16G16B16A16_FLOAT
+        && !desc

+fn texture(
+    device: &ID3D12Device,
+    desc: &D3D12_RESOURCE_DESC,
+    state: D3D12_RESOURCE_STATES,
+) -> Option<ID3D12Resource> {
+    let mut resource = None;
+    // SAFETY: plain committed resource on the host device.
+    unsafe {
+        device
+            .CreateCommittedResource(
+                &D3D12_HEAP_PROPERTIES {
+                    Type: D3D12_HEAP_TYPE_DEFAULT,
+                    ..Default::default()
+                },
+                D3D12_HEAP_FLAG_NONE,
+                desc,
+                state,
+                None,
+                &mut resource,
+            )
+            .ok()?;
+    }

+pub(crate) fn before(list: &HostList, title: This) -> Option<()> {
+    let host = host()?;
+    let color = crate::resource::made_copy(title)?;
+    // SAFETY: the Arc holds the color resource.
+    let mut desc = unsafe { color.texture.GetDesc() };
+    if !is_color(desc) {
+        return None;
+    }
+    desc.Flags = D3D12_RESOURCE_FLAG_NONE;
+    let texture = texture(&host.device, &desc, D3D12_RESOURCE_STATE_COPY_DEST)?;
+    list.keep(&texture);
+    crate::list::ensure_state(list, title, D3D12_RESOURCE_STATE_COPY_SOURCE.0 as u32);
+    // SAFETY: open list, matching descriptions and live resources.
+    unsafe { list.list.CopyResource(&texture, &color.texture) };
+    crate::list::ensure_state(list, title, D3D12_RESOURCE_STATE_RENDER_TARGET.0 as u32);
+    *list.dof_before() = Some(Before {
+        texture,
+        title: title as usize,
+        serial: color.serial,
+        epoch: list.epoch(),
+        frame: crate::queue::presented() + 1,
+    });

+fn pipeline(device: &ID3D12Device) -> Option<&'static Pipeline> {
+    fn create(device: &ID3D12Device) -> windows::core::Result<Pipeline> {
+        let code = crate::host::compile(SOURCE, s!("main"), s!("cs_6_0"))?;
+        let mut support = D3D12_FEATURE_DATA_FORMAT_SUPPORT {
+            Format: DXGI_FORMAT_R16_FLOAT,
+            ..Default::default()
+        };
+        // SAFETY: plain device capability query and pipeline creation.
+        unsafe {
+            device.CheckFeatureSupport(
+                D3D12_FEATURE_FORMAT_SUPPORT,
+                std::ptr::from_mut(&mut support).cast(),
+                size_of_val(&support) as u32,
+            )?;
+            if !support
+                .Support2
+                .contains(D3D12_FORMAT_SUPPORT2_UAV_TYPED_STORE)
+            {
+                return Err(windows::core::Error::empty());
+            }
+            let root: Option<ID3D12RootSignature> = Some(device.CreateRootSignature(0, &code)?);
+            let pso = device.CreateComputePipelineState(&D3D12_COMPUTE_PIPELINE_STATE_DESC {
+                pRootSignature: std::mem::transmute_copy(&root),
+                CS: D3D12_SHADER_BYTECODE {
+                    pShaderBytecode: code.as_ptr().cast(),
+                    BytecodeLength: code.len(),
+                },
+                ..Default::default()
+            });
+            Ok(Pipeline {
+                root: root.ok_or_else(windows::core::Error::empty)?,
+                pso: pso?,
+            })
+        }
+    }
+    static PIPELINE: OnceLock<Option<Pipeline>> = OnceLock::new();
+    PIPELINE
+        .get_or_init(|| {
+            create(device)
+                .inspect_err(|error| tracing::warn!(%error, "DOF guide pipeline unavailable"))
+                .ok()
+        })
+        .as_ref()

+pub(crate) fn produce(list: &HostList, title: This) -> Option<ID3D12Resource> {
+    if !enabled() {
+        return None;
+    }
+    let Some(before) = list.dof_before().take() else {
+        return None;
+    };
+    if title.is_null() || !crate::resource::is_resource(title) {
+        return None;
+    }
+    let color = crate::resource::made_copy(title)?;
+    let frame = crate::queue::presented() + 1;
+    if !same_frame(
+        [
+            before.title as u64,
+            before.serial,
+            before.epoch,
+            before.frame,
+        ],
+        [title as u64, color.serial, list.epoch(), frame],
+    ) || crate::resource::alias_of(title).is_some()
+        || crate::texture::title_written(&color.texture)
+    {
+        tracing::debug!(frame, "DOF guide withheld for mismatched or dirty handoff");
+        return None;
+    }
+    let host = host()?;
+    let device = &host.device;
+    let pipeline = pipeline(device)?;
+    // SAFETY: matched to the resource held during the pre-composite copy.
+    let mut desc = unsafe { color.texture.GetDesc() };
+    desc.Format = DXGI_FORMAT_R16_FLOAT;
+    desc.Flags = D3D12_RESOURCE_FLAG_ALLOW_UNORDERED_ACCESS;
+    let output = texture(device, &desc, D3D12_RESOURCE_STATE_UNORDERED_ACCESS)?;
+    // SAFETY: plain heap creation.
+    let heap: ID3D12DescriptorHeap = unsafe {
+        device
+            .CreateDescriptorHeap(&D3D12_DESCRIPTOR_HEAP_DESC {
+                Type: D3D12_DESCRIPTOR_HEAP_TYPE_CBV_SRV_UAV,
+                NumDescriptors: 3,
+                Flags: D3D12_DESCRIPTOR_HEAP_FLAG_SHADER_VISIBLE,
+                NodeMask: 0,
+            })
+            .ok()?
+    };
+    list.keep(&output);
+    list.keep(&heap);
+    // SAFETY: the heap has three descriptors and the list holds all three textures.
+    unsafe {
+        let stride = device.GetDescriptorHandleIncrementSize(D3D12_DESCRIPTOR_HEAP_TYPE_CBV_SRV_UAV)
+            as usize;
+        let mut cpu = heap.GetCPUDescriptorHandleForHeapStart();
+        device.CreateShaderResourceView(&before.texture, None, cpu);
+        cpu.ptr += stride;
+        device.CreateShaderResourceView(&color.texture, None, cpu);
+        cpu.ptr += stride;
+        device.CreateUnorderedAccessView(&output, None::<&ID3D12Resource>, None, cpu);
+    }
+    crate::list::ensure_state(list, title, INPUT.0 as u32);
+    // SAFETY: open list, held resources, exact dimensions divisible by 16.
+    unsafe {
+        list.list.barriers(&[transition(
+            &before.texture,
+            D3D12_RESOURCE_STATE_COPY_DEST,
+            INPUT,
+        )]);
+        list.list.SetDescriptorHeaps(&[Some(heap.clone())]);
+        list.list.SetComputeRootSignature(&pipeline.root);
+        list.list.SetPipelineState(&pipeline.pso);
+        list.list
+            .SetComputeRootDescriptorTable(0, heap.GetGPUDescriptorHandleForHeapStart());
+        list.list.Dispatch(WIDTH / 16, HEIGHT / 16, 1);
+        list.list.barriers(&[transition(
+            &output,
+            D3D12_RESOURCE_STATE_UNORDERED_ACCESS,
+            INPUT,
+        )]);
+    }
+    capture_check(list, &output, frame);
+    static RECORDED: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
+    let recorded = RECORDED.fetch_add(1, std::sync::atomic::Ordering::Relaxed) + 1;
+    if recorded <= 4 || recorded.is_multiple_of(256) {
+        tracing::info!(
+            frame,
+            serial = color.serial,
+            recorded,
+            "DOF guide recorded before FidelityFX"
+        );
+    }
+    Some(output)
+}

+fn capture_check(list: &HostList, source: &ID3D12Resource, frame: u64) {
+    static FRAME: OnceLock<Option<u64>> = OnceLock::new();
+    static CHECKED: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
+    let first = *FRAME.get_or_init(|| std::env::var("VICTORY_DOF_CHECK_FRAME").ok()?.parse().ok());
+    if first.is_none_or(|first| frame < first)
+        || CHECKED.swap(true, std::sync::atomic::Ordering::Relaxed)
+    {
+        return;
+    }
+    let Some(buffer) = host().and_then(|h| h.read_back_buffer(u64::from(PITCH * HEIGHT))) else {
+        return;
+    };
+    // SAFETY: copy locations borrow the held resources through submission.
+    unsafe {
+        let from = D3D12_TEXTURE_COPY_LOCATION {
+            pResource: std::mem::transmute_copy(source),
+            Type: D3D12_TEXTURE_COPY_TYPE_SUBRESOURCE_INDEX,
+            Anonymous: D3D12_TEXTURE_COPY_LOCATION_0 {
+                SubresourceIndex: 0,
+            },
+        };
+        let to = D3D12_TEXTURE_COPY_LOCATION {
+            pResource: std::mem::transmute_copy(&buffer),
+            Type: D3D12_TEXTURE_COPY_TYPE_PLACED_FOOTPRINT,
+            Anonymous: D3D12_TEXTURE_COPY_LOCATION_0 {
+                PlacedFootprint: D3D12_PLACED_SUBRESOURCE_FOOTPRINT {
+                    Offset: 0,
+                    Footprint: D3D12_SUBRESOURCE_FOOTPRINT {
+                        Format: DXGI_FORMAT_R16_FLOAT,
+                        Width: WIDTH,
+                        Height: HEIGHT,
+                        Depth: 1,
+                        RowPitch: PITCH,
+                    },
+                },
+            },
+        };
+        list.list
+            .barriers(&[transition(source, INPUT, D3D12_RESOURCE_STATE_COPY_SOURCE)]);
+        list.list.CopyTextureRegion(&to, 0, 0, 0, &from, None);
+        list.list
+            .barriers(&[transition(source, D3D12_RESOURCE_STATE_COPY_SOURCE, INPUT)]);
+    }
+    list.dof_readback(Readback { buffer, frame });
+}

+pub(crate) struct Readback {
+    buffer: ID3D12Resource,
+    frame: u64,
+}
+
+impl Readback {
+    pub(crate) fn check(&self) {
+        let bytes = (PITCH * HEIGHT) as usize;
+        let mut ptr = std::ptr::null_mut();
+        // SAFETY: called after the host submission fence; the whole footprint was copied.
+        let map = unsafe {
+            self.buffer.Map(
+                0,
+                Some(&D3D12_RANGE {
+                    Begin: 0,
+                    End: bytes,
+                }),
+                Some(&mut ptr),
+            )
+        };
+        if map.is_err() || ptr.is_null() {
+            tracing::warn!(frame = self.frame, "DOF guide readback failed");
+            return;
+        }
+        let [mut zero, mut positive, mut negative, mut non_finite] = [0; 4];
+        let mut max_abs_half = 0;
+        // SAFETY: successful Map covers the footprint.
+        let data = unsafe { std::slice::from_raw_parts(ptr.cast::<u8>(), bytes) };
+        for row in data.chunks_exact(PITCH as usize) {
+            for p in row[..(WIDTH * 2) as usize].chunks_exact(2) {
+                let bits = u16::from_le_bytes([p[0], p[1]]);
+                let absolute = bits & 0x7fff;
+                if absolute >= 0x7c00 {
+                    non_finite += 1;
+                } else if absolute == 0 {
+                    zero += 1;
+                } else if bits == absolute {
+                    positive += 1;
+                } else {
+                    negative += 1;
+                }
+                max_abs_half = max_abs_half.max(absolute);
+            }
+        }
+        // SAFETY: the readback is still mapped; no CPU writes.
+        unsafe {
+            self.buffer
+                .Unmap(0, Some(&D3D12_RANGE { Begin: 0, End: 0 }))
+        };
+        tracing::info!(
+            frame = self.frame,
+            zero,
+            positive,
+            negative,
+            non_finite,
+            max_abs_half,
+            "DOF guide GPU readback"
+        );
+    }
+}

+//! Opt-in host pipeline statistics for the title's reflection and lighting passes.
+
+use std::sync::{
+    Mutex, OnceLock,
+    atomic::{AtomicU32, Ordering::Relaxed},
+};
+
+use windows::Win32::Graphics::Direct3D12::*;
+
+use super::{HostList, host};
+
+const CAPACITY: u32 = 128;
+const BYTES: usize = size_of::<D3D12_QUERY_DATA_PIPELINE_STATISTICS>();
+const HASHES: &[&str] = &[
+    "34992f2b65a789a5", // Alpha temporal
+    "71883e40620a567a", // Opaque temporal and frame statistics
+    "dce085c07b9414b6", // Opaque bilateral
+    "6fad2980c2774c90", // Opaque history
+    "63381c3250b287fa",
+    "ff6fdf23ce0b9048", // Spatial variants
+    "7168dd781ca7efb9",
+    "d2b169b82f67ec5b",
+    "6d7fe7745ab9c6d8", // Ray prepass and trace
+    "2dc364df5d3278b8",
+    "940c2635dc1312f4",
+    "34fb51b8a0312e82", // Indirect diffuse
+    "577248728544d74b", // Apply combined deferred lighting
+];
+
+struct Queries {
+    heap: ID3D12QueryHeap,
+    readback: ID3D12Resource,
+    mapped: usize,
+    next: AtomicU32,
+    pending: Mutex<Vec<Pending>>,
+}
+
+#[derive(Clone, Copy)]
+pub(crate) struct Sample {
+    slot: u32,
+    recorded_frame: u64,
+    shader: &'static str,
+    max_commands: u32,
+    sent: bool,
+}
+
+struct Pending {
+    sample: Sample,
+    fence: u64,
+    frame: u64,
+    submission: u64,
+    list_index: usize,
+    list_epoch: u64,
+}
+
+static QUERIES: OnceLock<Option<Queries>> = OnceLock::new();
+
+fn capture_frame() -> Option<u64> {
+    static FRAME: OnceLock<Option<u64>> = OnceLock::new();
+    *FRAME.get_or_init(|| {
+        let value = std::env::var_os("VICTORY_PASS_STATS_FRAME")?;
+        let frame = value
+            .to_str()
+            .and_then(|s| s.parse().ok())
+            .filter(|&n| n != 0);
+        if frame.is_none() {
+            tracing::warn!(
+                "pass statistics unavailable: expected a positive VICTORY_PASS_STATS_FRAME"
+            );
+        }
+        frame
+    })
+}
+
+fn frame() -> u64 {
+    crate::queue::presented() + 1
+}
+
+fn create() -> Option<Queries> {
+    let host = host()?;
+    let mut heap = None;
+    let mut mapped = std::ptr::null_mut();
+    // SAFETY: plain creation and mapping. Dedicated lifetime storage, separate from guest queries.
+    unsafe {
+        host.device
+            .CreateQueryHeap(
+                &D3D12_QUERY_HEAP_DESC {
+                    Type: D3D12_QUERY_HEAP_TYPE_PIPELINE_STATISTICS,
+                    Count: CAPACITY,
+                    NodeMask: 0,
+                },
+                &mut heap,
+            )
+            .ok()?;
+        let readback = host.read_back_buffer((BYTES * CAPACITY as usize) as u64)?;
+        readback.Map(0, None, Some(&mut mapped)).ok()?;
+        if mapped.is_null() {
+            return None;
+        }
+        std::ptr::write_bytes(mapped.cast::<u8>(), 0xff, BYTES * CAPACITY as usize);
+        Some(Queries {
+            heap: heap?,
+            readback,
+            mapped: mapped as usize,
+            next: AtomicU32::new(0),
+            pending: Mutex::new(Vec::new()),
+        })
+    }
+}

+pub(crate) fn before(list: &HostList, shader: Option<&str>, max_commands: u32) -> Option<Sample> {
+    let target = capture_frame()?;
+    let recorded_frame = frame();
+    if !(target.saturating_sub(2)..=target).contains(&recorded_frame) {
+        return None;
+    }
+    let shader = HASHES.iter().copied().find(|&hash| Some(hash) == shader)?;
+    let queries = QUERIES
+        .get_or_init(|| {
+            let queries = create();
+            if queries.is_none() {
+                tracing::warn!(
+                    target,
+                    "pass statistics unavailable: could not create host query heap/readback"
+                );
+            }
+            queries
+        })
+        .as_ref()?;
+    let slot = queries.next.fetch_add(1, Relaxed);
+    if slot >= CAPACITY {
+        if slot == CAPACITY {
+            tracing::warn!(
+                target,
+                CAPACITY,
+                "pass statistics incomplete: slot limit reached"
+            );
+        }
+        return None;
+    }
+    // SAFETY: the graphics list is open. This slot belongs to this dispatch and is never reused.
+    unsafe {
+        list.list
+            .BeginQuery(&queries.heap, D3D12_QUERY_TYPE_PIPELINE_STATISTICS, slot)
+    };
+    Some(Sample {
+        slot,
+        recorded_frame,
+        shader,
+        max_commands,
+        sent: false,
+    })
+}
+
+pub(crate) fn after(list: &HostList, sample: Option<Sample>) {
+    let (Some(sample), Some(queries)) = (sample, QUERIES.get().and_then(Option::as_ref)) else {
+        return;
+    };
+    // SAFETY: paired to before() on this same list, with a disjoint readback range per slot.
+    unsafe {
+        list.list.EndQuery(
+            &queries.heap,
+            D3D12_QUERY_TYPE_PIPELINE_STATISTICS,
+            sample.slot,
+        );
+        list.list.ResolveQueryData(
+            &queries.heap,
+            D3D12_QUERY_TYPE_PIPELINE_STATISTICS,
+            sample.slot,
+            1,
+            &queries.readback,
+            (BYTES * sample.slot as usize) as u64,
+        );
+    }
+    list.pass_stats
+        .lock()
+        .unwrap_or_else(|e| e.into_inner())
+        .push(sample);
+}

+/// Associate the recorded samples with the actual host submission and its trailing fence.
+pub(crate) fn submitted(frame: u64, submission: u64, runs: &[&HostList], fence: u64) {
+    let Some(queries) = QUERIES.get().and_then(Option::as_ref) else {
+        return;
+    };
+    let mut pending = queries.pending.lock().unwrap_or_else(|e| e.into_inner());
+    for (list_index, list) in runs.iter().enumerate() {
+        let list_epoch = list.epoch();
+        for sample in list
+            .pass_stats
+            .lock()
+            .unwrap_or_else(|e| e.into_inner())
+            .iter_mut()
+        {
+            if sample.sent {
+                pending.retain(|p| p.sample.slot != sample.slot);
+                tracing::warn!(
+                    frame,
+                    submission,
+                    list_index,
+                    list_epoch,
+                    slot = sample.slot,
+                    "pass statistics incomplete: a sampled list ran again"
+                );
+                continue;
+            }
+            sample.sent = true;
+            pending.push(Pending {
+                sample: *sample,
+                fence,
+                frame,
+                submission,
+                list_index,
+                list_epoch,
+            });
+        }
+    }
+}

+/// Called under the host submission lock. The fence orders the resolved readback.
+pub(crate) fn report_completed(fence: &ID3D12Fence) {
+    let Some(queries) = QUERIES.get().and_then(Option::as_ref) else {
+        return;
+    };
+    let mut pending = queries.pending.lock().unwrap_or_else(|e| e.into_inner());
+    if pending.is_empty() {
+        return;
+    }
+    // SAFETY: a plain query of the host's queue fence.
+    let completed = unsafe { fence.GetCompletedValue() };
+    if completed == u64::MAX {
+        pending.clear();
+        tracing::warn!("pass statistics unavailable: host device removed");
+        return;
+    }
+    pending.retain(|p| {
+        if p.fence > completed {
+            return true;
+        }
+        let Sample {
+            slot,
+            recorded_frame,
+            shader,
+            max_commands,
+            ..
+        } = p.sample;
+        // SAFETY: private mapped result, below the signalled fence and within its slot.
+        let cs_invocations = unsafe {
+            let at = (queries.mapped + BYTES * slot as usize)
+                as *const D3D12_QUERY_DATA_PIPELINE_STATISTICS;
+            std::ptr::addr_of!((*at).CSInvocations).read_volatile()
+        };
+        let (frame, submission, list_index, list_epoch, fence) =
+            (p.frame, p.submission, p.list_index, p.list_epoch, p.fence);
+        if cs_invocations == u64::MAX {
+            tracing::warn!(
+                frame,
+                submission,
+                list_index,
+                list_epoch,
+                slot,
+                recorded_frame,
+                shader,
+                fence,
+                "pass statistics incomplete: no resolved value at retired fence"
+            );
+        } else {
+            tracing::info!(
+                frame,
+                submission,
+                list_index,
+                list_epoch,
+                slot,
+                recorded_frame,
+                shader,
+                max_commands,
+                fence,
+                cs_invocations,
+                "retired pass statistics"
+            );
+        }
+        false
+    });
