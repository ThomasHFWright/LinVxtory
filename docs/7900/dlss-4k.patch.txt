+pub(super) enum Dispatch {

+pub(super) fn dispatch(handle: usize, info: &DispatchUpscale) -> Dispatch {
+    dispatch_in(&ACTIVE, handle, |active| {
+        if is_owned(handle)
+            && (info.list.is_null()
+                || !supported(info, UHD)
+                || info.settings[0] & 0xff != 0 // This provider does not apply FFX sharpening.
+                || [0, 1, 2, 6]
+                    .iter()
+                    .any(|&i| info.resources[i].resource.is_null()))
+        {
+            return Err("invalid UHD command list or dispatch descriptor");
+        }
+        external::record(info.list.cast::<Object>(), |list| {
+            record(list, info, active)
+        })
+        .unwrap_or(Err("command list unavailable"))
+    })
+}

+fn dispatch_in(
+    context: &Context,
+    handle: usize,
+    record: impl FnOnce(&mut Active) -> Result<(), &'static str>,
+) -> Dispatch {
+    let unavailable = || {
+        if is_owned(handle) {
+            Dispatch::OwnedFailed
+        } else {
+            Dispatch::Stock
+        }
+    };
+    let Some(active) = context.get() else {
+        return unavailable();
+    };
+    let mut active = active.lock().unwrap_or_else(|e| e.into_inner());
+    if active.context == 0 || active.context != handle || active.failed {
+        return unavailable();
+    }
+    match record(&mut active) {
+        Ok(()) => {
+            active.replaced_history = true;
+            let count = REPLACED.fetch_add(1, Ordering::Relaxed) + 1;
+            if count.is_power_of_two() {
+                tracing::info!(count, render = ?INPUT, output = ?active.preset.output, "Series S DLSS replaced stock upscale");
+            }
+            Dispatch::Replaced
+        }
+        Err(reason) => {
+            active.failed = true;
+            if is_owned(handle) {
+                tracing::error!(reason, "NGX-owned FidelityFX UHD dispatch failed");
+                return Dispatch::OwnedFailed;
+            }
+            tracing::warn!(reason, "Series S DLSS fell back to stock upscale");
+            if active.replaced_history {
+                Dispatch::ResetStock
+            } else {
+                Dispatch::Stock
+            }
+        }
+    }
+}

+        && dispatch_output(info.render_size, info.upscale_size, INPUT, preset.output)
+            == Some(preset.output)
+        && info.resources[3..6].iter().all(|r| r.resource.is_null())
+        && info.settings[9] == 0
+        && exposure.is_finite()
+        && exposure > 0.0
+        && info
+            .jitter
+            .iter()
+            .chain(&info.motion_scale)
+            .all(|v| v.is_finite())

+fn record(
+    list: &external::Recording<'_>,
+    info: &DispatchUpscale,
+    active: &mut Active,
+) -> Result<(), &'static str> {
+    let preset = active.preset;
+    if !supported(info, preset) {
+        return Err("upscale descriptor outside the tested preset");
+    }
+    let input = |role: usize| {
+        let title = info
+            .resources
+            .get(role)
+            .filter(|r| !r.resource.is_null())
+            .ok_or("missing required input")?
+            .resource
+            .cast::<Object>();
+        list.prepare_input(title)
+            .ok_or("input preparation failed")?;
+        let host = texture(
+            list,
+            title,
+            D3D12_RESOURCE_STATE_NON_PIXEL_SHADER_RESOURCE,
+            INPUT,
+        )?;
+        if is_owned(active.context) {
+            // SAFETY: the list holds this resource through the call.
+            let desc = unsafe { host.GetDesc() };
+            if desc.Dimension != D3D12_RESOURCE_DIMENSION_TEXTURE2D
+                || desc.Width != u64::from(INPUT[0])
+                || desc.Height != INPUT[1]
+            {
+                return Err("expected an exact 1280x720 DLSS input allocation");
+            }
+        }
+        Ok(host)
+    };
+    let color = input(0)?;
+    let depth = input(1)?;
+    let motion = input(2)?;
+    let output = info.resources[6].resource.cast::<Object>();
+    let output_host = texture(
+        list,
+        output,
+        D3D12_RESOURCE_STATE_UNORDERED_ACCESS,
+        preset.output,
+    )?;
+    // SAFETY: all four resources remain held by the host command list.
+    let output_desc = unsafe { output_host.GetDesc() };
+    if preset.output == UHD.output
+        && (output_desc.Dimension != D3D12_RESOURCE_DIMENSION_TEXTURE2D
+            || output_desc.Width != u64::from(preset.output[0])
+            || output_desc.Height != preset.output[1])
+    {
+        return Err("expected an exact 3840x2160 DLSS output allocation");
+    }
+    let output_format = output_desc.Format;
+    let device = external::host_device().ok_or("no host device")?;
+    let support = &mut D3D12_FEATURE_DATA_FORMAT_SUPPORT {
+        Format: output_format,
+        ..Default::default()
+    };
+    // SAFETY: plain format capability query, using exactly the declared structure.
+    let supports_store = unsafe {
+        device.CheckFeatureSupport(
+            D3D12_FEATURE_FORMAT_SUPPORT,
+            std::ptr::from_mut(support).cast(),
+            size_of_val(support) as u32,
+        )
+    }
+    .is_ok()
+        && support
+            .Support2
+            .contains(D3D12_FORMAT_SUPPORT2_UAV_TYPED_STORE);
+    if output_format.0 != 10 || !supports_store {
+        return Err("upscale output must have a typed RGBA16F UAV backing");
+    }
+    let commands = list.commands();
+    if active.provider.is_none() {
+        active.provider =
+            Some(init(&device, commands, preset).ok_or("NGX init or feature creation failed")?);
+    }
+    let ngx = active.provider.as_mut().unwrap();
+    if ngx.feature.is_null() {
+        ngx.feature = create_feature(commands, preset, ngx.params)
+            .ok_or("NGX Ultra Performance feature creation failed")?;
+    }
+    let p = ngx.params;
+    // SAFETY: this list, its four held resources and the live NGX feature all use the same device.
+    let result = unsafe {
+        NVSDK_NGX_Parameter_SetD3d12Resource(p, name!("Color"), color.as_raw());
+        NVSDK_NGX_Parameter_SetD3d12Resource(p, name!("Depth"), depth.as_raw());
+        NVSDK_NGX_Parameter_SetD3d12Resource(p, name!("MotionVectors"), motion.as_raw());
+        NVSDK_NGX_Parameter_SetD3d12Resource(p, name!("Output"), output_host.as_raw());
+        NVSDK_NGX_Parameter_SetF(p, name!("Jitter.Offset.X"), info.jitter[0]);
+        NVSDK_NGX_Parameter_SetF(p, name!("Jitter.Offset.Y"), info.jitter[1]);
+        NVSDK_NGX_Parameter_SetF(p, name!("MV.Scale.X"), info.motion_scale[0]);
+        NVSDK_NGX_Parameter_SetF(p, name!("MV.Scale.Y"), info.motion_scale[1]);
+        NVSDK_NGX_Parameter_SetI(
+            p,
+            name!("Reset"),
+            i32::from(!active.replaced_history || info.settings[4] & 0xff != 0),
+        );
+        NVSDK_NGX_Parameter_SetF(
+            p,
+            name!("DLSS.Pre.Exposure"),
+            f32::from_bits(info.settings[3]),
+        );
+        NVSDK_NGX_Parameter_SetF(
+            p,
+            name!("FrameTimeDeltaInMsec"),
+            f32::from_bits(info.settings[2]),
+        );
+        NVSDK_NGX_Parameter_SetUI(p, name!("DLSS.Render.Subrect.Dimensions.Width"), INPUT[0]);
+        NVSDK_NGX_Parameter_SetUI(p, name!("DLSS.Render.Subrect.Dimensions.Height"), INPUT[1]);
+        NVSDK_NGX_D3D12_EvaluateFeature_C(commands.as_raw(), ngx.feature, p, std::ptr::null())
+    };
+    if result != SUCCESS {
+        tracing::warn!(result, "NGX did not evaluate the upscale");
+        return Err("NGX evaluation failed");
+    }
+    external::output_written(output);
+    Ok(())
+}

+fn texture(
+    list: &external::Recording<'_>,
+    title: *const Object,
+    state: D3D12_RESOURCE_STATES,
+    size: [u32; 2],
+) -> Result<ID3D12Resource, &'static str> {
+    if title.is_null() {
+        return Err("missing upscale resource");
+    }
+    let (host, effective) = list
+        .texture(title, state.0 as u32)
+        .ok_or("missing upscale backing")?;
+    // SAFETY: GetDesc queries this live resource by value.
+    let desc = unsafe { host.GetDesc() };
+    if effective != state.0 as u32 || desc.Width < u64::from(size[0]) || desc.Height < size[1] {
+        return Err("upscale backing has incompatible state or dimensions");
+    }
+    Ok(host)
+}

+fn initialize_nvapi() -> bool {
+    static NVAPI: OnceLock<bool> = OnceLock::new();
+    *NVAPI.get_or_init(|| {
+        use windows_sys::Win32::System::LibraryLoader::{GetProcAddress, LoadLibraryW};
+        let name = windows::core::w!("nvapi64.dll");
+        // SAFETY: the probe stages dxvk-nvapi beside its executable.
+        unsafe {
+            let module = LoadLibraryW(name.as_ptr());
+            if module.is_null() {
+                return false;
+            }
+            let Some(symbol) = GetProcAddress(module, c"nvapi_QueryInterface".as_ptr().cast())
+            else {
+                return false;
+            };
+            let query =
+                std::mem::transmute::<_, unsafe extern "C" fn(u32) -> *const c_void>(symbol);
+            let init = query(0x0150_e828); // NvAPI_Initialize, NVIDIA nvapi_interface.h.
+            if init.is_null() {
+                return false;
+            }
+            let start = std::mem::transmute::<_, unsafe extern "C" fn() -> i32>(init);
+            let status = start();
+            eprintln!("Series S DLSS NvAPI initialize status={status}");
+            status == 0
+        }
+    })
+}

+unsafe extern "C" fn ngx_log(message: *const c_char, _: c_int, _: c_int) {
+    static LOGGED: AtomicUsize = AtomicUsize::new(0);
+    if !message.is_null() && LOGGED.fetch_add(1, Ordering::Relaxed) < 48 {
+        // SAFETY: the SDK supplies a NUL-terminated message for this call.
+        let message = unsafe { std::ffi::CStr::from_ptr(message) };
+        eprintln!("NGX: {}", message.to_string_lossy().trim_end());
+    }

+fn capabilities(device: &ID3D12Device) -> Option<*mut c_void> {
+    use std::os::windows::ffi::OsStrExt;
+
+    let dir = std::env::current_exe().ok()?.parent()?.to_path_buf();
+    if !dir.join("nvngx_dlss.dll").is_file() {
+        return None;
+    }
+    let data = std::env::temp_dir().join("victory-ngx");
+    std::fs::create_dir_all(&data).ok()?;
+    let dir_w: Vec<u16> = dir.as_os_str().encode_wide().chain([0]).collect();
+    let data_w: Vec<u16> = data.as_os_str().encode_wide().chain([0]).collect();
+    let paths = [dir_w.as_ptr()];
+    let info = CommonInfo {
+        paths: PathList {
+            path: paths.as_ptr(),
+            length: 1,
+        },
+        internal: std::ptr::null(),
+        logging: Logging {
+            callback: ngx_log as *const c_void,
+            minimum: 2,
+            disable_other_sinks: false,
+        },
+    };
+    let mut params = std::ptr::null_mut();
+    // SAFETY: live device and command list, strings and out pointers for the SDK.
+    unsafe {
+        let result = NVSDK_NGX_D3D12_Init_with_ProjectID(
+            c"5e2f9a3c-1b7d-4c8e-9a6f-0d3b2c1e4f57".as_ptr(),
+            ENGINE_CUSTOM,
+            c"1.0".as_ptr(),
+            data_w.as_ptr(),
+            device.as_raw(),
+            &info,
+            VERSION_API,
+        );
+        if result != SUCCESS {
+            eprintln!(
+                "Series S DLSS NGX initialization failed ({:#010x})",
+                result as u32
+            );
+            return None;
+        }
+        if NVSDK_NGX_D3D12_GetCapabilityParameters(&mut params) != SUCCESS || params.is_null() {
+            return None;
+        }
+        let mut available = 0;
+        if NVSDK_NGX_Parameter_GetI(params, name!("SuperSampling.Available"), &mut available)
+            != SUCCESS
+            || available == 0
+        {
+            return None;
+        }
+    }
+    Some(params)
+}

+fn init(
+    device: &ID3D12Device,
+    commands: &ID3D12GraphicsCommandList,
+    preset: Preset,
+) -> Option<Ngx> {
+    if !initialize_nvapi() {
+        return None;
+    }
+    let params = capabilities(device)?;
+    let feature = create_feature(commands, preset, params)?;
+    Some(Ngx { params, feature })

+fn create_feature(
+    commands: &ID3D12GraphicsCommandList,
+    preset: Preset,
+    params: *mut c_void,
+) -> Option<*mut c_void> {
+    let mut feature = std::ptr::null_mut();
+    // SAFETY: live SDK parameters and an open command list.
+    unsafe {
+        for (parameter, value) in [
+            (name!("CreationNodeMask"), 1),
+            (name!("VisibilityNodeMask"), 1),
+            (name!("Width"), INPUT[0]),
+            (name!("Height"), INPUT[1]),
+            (name!("OutWidth"), preset.output[0]),
+            (name!("OutHeight"), preset.output[1]),
+        ] {
+            NVSDK_NGX_Parameter_SetUI(params, parameter, value);
+        }
+        NVSDK_NGX_Parameter_SetI(params, name!("PerfQualityValue"), preset.quality);
+        NVSDK_NGX_Parameter_SetI(
+            params,
+            name!("DLSS.Feature.Create.Flags"),
+            FLAG_HDR | FLAG_MV_LOW_RES | FLAG_DEPTH_INVERTED | FLAG_AUTO_EXPOSURE,
+        );
+        NVSDK_NGX_Parameter_SetI(params, name!("DLSS.Enable.Output.Subrects"), 0);
+        let result = NVSDK_NGX_D3D12_CreateFeature(
+            commands.as_raw(),
+            FEATURE_SUPER_SAMPLING,
+            params,
+            &mut feature,
+        );
+        if result != SUCCESS || feature.is_null() {
+            tracing::warn!(result, "Series S DLSS NGX feature creation failed");
+            return None;
+        }
+    }
+    Some(feature)
+}

+            }
+            #[cfg(feature = "dlss")]
+            "ffxConfigure" if super::ngx::owns_uhd() => configure_wrapper(address),
+            "ffxQuery" => {
+                let saved = QUERY.get_or_init(|| {
+                    tracing::info!("console FidelityFX named query import observed");
+                    // SAFETY: original, named console export with the known FFX ABI.
+                    unsafe { std::mem::transmute::<usize, Call>(address) }
+                });
+                (*saved as usize == address).then_some(observe_query as *const () as usize)
+            }
+            "ffxDispatch" => {
+                // SAFETY: original, named console export with the known FFX ABI.
+                let saved =
+                    DISPATCH.get_or_init(|| unsafe { std::mem::transmute::<usize, Call>(address) });
+                (*saved as usize == address).then_some(observe_dispatch as *const () as usize)
+            }
+            _ => None,
+        };
+        #[cfg(feature = "dlss")]
+        if super::ngx::owns_uhd()
+            && wrapped.is_none()
+            && matches!(
+                name.as_str(),
+                "ffxCreateContext"
+                    | "ffxDestroyContext"
+                    | "ffxQuery"
+                    | "ffxDispatch"
+                    | "ffxConfigure"
+            )
+        {
+            return Resolution::Missing;
+        }
+        wrapped.map_or(original, |address| Resolution::Found {
+            address,
+            provenance: Provenance::Title,
+        })
+    }
+}

+/// The caller must use a layout that accepts all bit patterns.
+unsafe fn snapshot<T: Copy>(source: *const T) -> Option<T> {
+    let mut data = MaybeUninit::<T>::uninit();
+    let mut read = 0;
+    // SAFETY: the OS checks the source mapping and writes at most the size of data.
+    let ok = unsafe {
+        ReadProcessMemory(
+            GetCurrentProcess(),
+            source.cast(),
+            data.as_mut_ptr().cast(),
+            size_of::<T>(),
+            &mut read,
+        )
+    };
+    // SAFETY: an exact copy into a layout that accepts all bit patterns.
+    (ok != 0 && read == size_of::<T>()).then(|| unsafe { data.assume_init() })
+}

+/// The upscaler's public ABI for querying the length of its jitter sequence.
+#[repr(C)]
+#[derive(Clone, Copy)]
+struct JitterPhaseCount {
+    header: Header,
+    render_width: u32,
+    display_width: u32,
+    out: *mut i32,

+const _: () = assert!(size_of::<JitterPhaseCount>() == 32);

+unsafe extern "C" fn observe_query(context: *mut *mut c_void, desc: *mut Header) -> u32 {
+    static CALLS: AtomicUsize = AtomicUsize::new(0);
+    // SAFETY: the original query gets the original pointers. Snapshots check mapped memory.
+    unsafe {
+        let last = GetLastError();
+        let info = snapshot(desc)
+            .filter(|h| h.kind == GET_JITTER_PHASE_COUNT && h.next.is_null())
+            .and_then(|_| snapshot(desc.cast::<JitterPhaseCount>()));
+        #[cfg(feature = "dlss")]
+        if super::ngx::owns_uhd()
+            && let Some(result) = owned::query(context, desc, info)
+        {
+            SetLastError(last);
+            return result;
+        }
+        SetLastError(last);
+        let result = QUERY.get().unwrap()(context, desc);
+        let last = GetLastError();
+        if let Some(info) = info {
+            let nth = CALLS.fetch_add(1, Ordering::Relaxed).wrapping_add(1);
+            if nth <= 8 || nth.is_power_of_two() {
+                let phases = (result == super::OK).then(|| snapshot(info.out)).flatten();
+                tracing::info!(
+                    nth,
+                    result,
+                    render_width = info.render_width,
+                    display_width = info.display_width,
+                    ?phases,
+                    "console FidelityFX queried jitter phase count"
+                );
+            }
+        }
+        SetLastError(last);
+        result
+    }
+}

+unsafe extern "C" fn observe_dispatch(context: *mut *mut c_void, desc: *mut Header) -> u32 {
+    static CALLS: AtomicUsize = AtomicUsize::new(0);
+    let nth = CALLS.fetch_add(1, Ordering::Relaxed).wrapping_add(1);
+    // SAFETY: the original ABI receives the same arguments and descriptor layouts accept arbitrary bits.
+    unsafe {
+        let last = GetLastError();
+        let sample = nth <= 8 || nth.is_power_of_two();
+        let info = if sample || replace_upscale() {
+            snapshot(desc)
+                .filter(|h| h.kind == DISPATCH_UPSCALE)
+                .and_then(|_| snapshot(desc.cast::<DispatchUpscale>()))
+        } else {
+            None
+        };
+        #[cfg(feature = "dlss")]
+        if super::ngx::owns_uhd()
+            && let Some(result) = owned::dispatch(context, info)
+        {
+            SetLastError(last);
+            return result;
+        }
+        SetLastError(last);
+        #[cfg(feature = "dlss")]
+        let mut fallback = None;
+        #[cfg(feature = "dlss")]
+        if replace_upscale()
+            && let Some(mut info) = info
+            && let Some(handle) = snapshot(context)
+        {
+            match super::ngx::dispatch(handle as usize, &info) {
+                super::ngx::Dispatch::Stock => {}
+                super::ngx::Dispatch::ResetStock => {
+                    info.settings[4] = (info.settings[4] & !0xff) | 1;
+                    fallback = Some(info);
+                }
+                super::ngx::Dispatch::Replaced => {
+                    SetLastError(last);
+                    return super::OK;
+                }
+                super::ngx::Dispatch::OwnedFailed => {
+                    SetLastError(last);
+                    return owned::RUNTIME_ERROR;
+                }
+            }
+        }
+        #[cfg(feature = "dlss")]
+        let desc = fallback.as_mut().map_or(desc, |info| &raw mut info.header);
+        SetLastError(last);
+        let result = DISPATCH.get().unwrap()(context, desc);
+        let last = GetLastError();
+        if let Some(info) = info.filter(|_| sample) {
+            let resources = info
+                .resources
+                .map(|r| (!r.resource.is_null()).then_some((r.description, r.state)));
+            tracing::info!(nth, result, render = ?info.render_size, output = ?info.upscale_size,
+                jitter = ?info.jitter, motion_scale = ?info.motion_scale,
+                pre_exposure = f32::from_bits(info.settings[3]), reset = (info.settings[4] & 0xff) != 0,
+                flags = info.settings[9], ?resources, "console FidelityFX dispatched upscaler");
+        }
+        SetLastError(last);
+        result
+    }
+}

+fn writable<T>(out: *mut T) -> bool {
+    if out.is_null() || !out.is_aligned() {
+        return false;
+    }
+    let mut region = MaybeUninit::<MEMORY_BASIC_INFORMATION>::uninit();
+    // SAFETY: VirtualQuery writes only to the local result.
+    let size = unsafe {
+        VirtualQuery(
+            out.cast(),
+            region.as_mut_ptr(),
+            size_of::<MEMORY_BASIC_INFORMATION>(),
+        )
+    };
+    if size != size_of::<MEMORY_BASIC_INFORMATION>() {
+        return false;
+    }
+    // SAFETY: the whole result was written.
+    let region = unsafe { region.assume_init() };
+    region.State == MEM_COMMIT
+        && matches!(
+            region.Protect,
+            PAGE_READWRITE | PAGE_WRITECOPY | PAGE_EXECUTE_READWRITE | PAGE_EXECUTE_WRITECOPY
+        )
+        && (out as usize)
+            .checked_sub(region.BaseAddress as usize)
+            .and_then(|offset| offset.checked_add(size_of::<T>()))
+            .is_some_and(|end| end <= region.RegionSize)
+}

+fn write<T: Copy>(out: *mut T, value: T) -> bool {
+    if !writable(out) {
+        return false;
+    }
+    let mut transferred = 0;
+    // SAFETY: the OS checks the destination for this in-process copy.
+    let ok = unsafe {
+        WriteProcessMemory(
+            GetCurrentProcess(),
+            out.cast(),
+            std::ptr::from_ref(&value).cast(),
+            size_of::<T>(),
+            &mut transferred,
+        )
+    };
+    ok != 0 && transferred == size_of::<T>()
+}

+fn overlaps<T, U>(out: *mut T, input: *const U) -> bool {
+    let (out, input) = (out as usize, input as usize);
+    out < input.saturating_add(size_of::<U>()) && input < out.saturating_add(size_of::<T>())
+}

+fn create_request(info: &CreateUpscale) -> bool {
+    if info.header.next.is_null() {
+        return true;
+    }
+    // The backend and the title's compiled FidelityFX version are metadata
+    // for the console provider. NGX obtains its device from the host.
+    // SAFETY: each snapshot checks the mapped chain link before reading it.
+    let first = unsafe { snapshot(info.header.next) };
+    let backend = first.filter(|link| link.kind == CONSOLE_BACKEND);
+    if backend.is_some_and(|link| link.next.is_null()) {
+        return true;
+    }
+    let second = backend.and_then(|link| unsafe { snapshot(link.next) });
+    let version = backend.zip(second).and_then(|(backend, link)| {
+        (link.kind == CONSOLE_VERSION && link.next.is_null())
+            .then(|| unsafe { snapshot(backend.next.cast::<ConsoleVersion>()) })
+            .flatten()
+    });
+    if version.is_some_and(|info| info.version == CONSOLE_API_4_1_1) {
+        return true;
+    }
+    tracing::warn!(
+        first_kind = first.map(|link| link.kind),
+        second_kind = second.map(|link| link.kind),
+        version = version.map(|info| info.version),
+        "NGX-owned FidelityFX UHD rejected an unknown creation extension"
+    );
+    false
+}

+pub(super) fn create(
+    context: *mut *mut c_void,
+    desc: *mut Header,
+    info: Option<CreateUpscale>,

+fn create_with(
+    context: *mut *mut c_void,
+    desc: *mut Header,
+    info: Option<CreateUpscale>,
+    open: impl FnOnce(*mut *mut c_void) -> u32,
+) -> Option<u32> {
+    let Some(current) = handle(context) else {
+        return Some(PARAMETER);
+    };
+    if ngx::is_owned(current) {
+        return Some(PARAMETER);
+    }
+    let info = info.filter(ngx::matches_uhd)?;
+    if current != 0 || overlaps(context, desc.cast::<CreateUpscale>()) || !writable(context) {
+        return Some(PARAMETER);
+    }
+    if !create_request(&info) {
+        return Some(UNKNOWN_TYPE);
+    }
+    let result = open(context);
+    tracing::info!(
+        result, flags = info.flags, max_render = ?info.max_render, max_output = ?info.max_output,
+        "NGX-owned FidelityFX UHD context creation"
+    );
+    Some(result)
+}

+pub(super) fn destroy(context: *mut *mut c_void) -> Option<u32> {
+    match handle(context) {
+        Some(handle) if ngx::is_owned(handle) => {
+            Some(ngx::destroy_uhd(|| write(context, null_mut())))
+        }
+        None | Some(0) => Some(PARAMETER),
+        _ => None,
+    }

+fn phase_dimensions(info: &JitterPhaseCount) -> bool {
+    (info.render_width, info.display_width) == (1280, 3840)
+}
+
+fn write_phase(context: *mut *mut c_void, desc: *mut Header, info: JitterPhaseCount) -> u32 {
+    if !phase_dimensions(&info)
+        || overlaps(info.out, context)
+        || overlaps(info.out, desc.cast::<JitterPhaseCount>())
+        || !write(info.out, UHD_PHASES)
+    {
+        return PARAMETER;
+    }
+    tracing::info!(
+        render_width = info.render_width,
+        display_width = info.display_width,
+        phases = UHD_PHASES,
+        "NGX-owned FidelityFX UHD jitter phase count"
+    );
+    OK
+}

+pub(super) fn query(
+    context: *mut *mut c_void,
+    desc: *mut Header,
+    info: Option<JitterPhaseCount>,
+) -> Option<u32> {
+    let Some(handle) = handle(context) else {
+        return Some(PARAMETER);
+    };
+    if ngx::is_owned(handle) {
+        // SAFETY: snapshot checks readable storage. A borrowed invalid query
+        // cannot escape to the stock provider with our token.
+        let invalid = if unsafe { snapshot(desc) }.is_some() {
+            UNKNOWN_TYPE
+        } else {
+            PARAMETER
+        };
+        return Some(info.map_or(invalid, |info| {
+            ngx::query_uhd(|| write_phase(context, desc, info))
+        }));
+    }
+    if handle == 0
+        && let Some(info) = info.filter(phase_dimensions)
+    {
+        return Some(ngx::query_uhd(|| write_phase(context, desc, info)));
+    }
+    None
+}

+pub(super) fn dispatch(context: *mut *mut c_void, info: Option<DispatchUpscale>) -> Option<u32> {
+    match handle(context) {
+        Some(handle) if ngx::is_owned(handle) => Some(match info {
+            Some(info) if ngx::dispatch(handle, &info) == ngx::Dispatch::Replaced => OK,
+            Some(_) => RUNTIME_ERROR,
+            None => PARAMETER,
+        }),
+        None | Some(0) => Some(PARAMETER),
+        _ => None,
+    }
