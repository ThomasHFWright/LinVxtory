+fn validate(
+    device: &ID3D12Device,
+    commands: &ID3D12GraphicsCommandList,
+    guides: &Guides<'_>,
+    frame: &Frame,
+) -> Result<(), Error> {
+    validate_temporal(frame)?;
+    // SAFETY: command type and device are immutable properties of a live COM object.
+    if unsafe { commands.GetType() } != D3D12_COMMAND_LIST_TYPE_DIRECT
+        || !on_device(commands, device)
+    {
+        return Err(Error::Invalid(
+            "RR requires a direct list on the host device",
+        ));
+    }
+    let motion_formats = [DXGI_FORMAT_R16G16_FLOAT, DXGI_FORMAT_R32G32_FLOAT];
+    let scalar_formats = [DXGI_FORMAT_R16_FLOAT, DXGI_FORMAT_R32_FLOAT];
+    let rgba = [DXGI_FORMAT_R16G16B16A16_FLOAT];
+    let albedo = [DXGI_FORMAT_R16G16B16A16_FLOAT, DXGI_FORMAT_R8G8B8A8_UNORM];
+    let (reflections, reflection_formats) = match guides.reflections {
+        Reflections::Motion(motion) => (motion, motion_formats),
+        Reflections::HitDistance {
+            distance,
+            world_to_view,
+            view_to_clip,
+        } => {
+            if !invertible(world_to_view) || !invertible(view_to_clip) {
+                return Err(Error::Invalid("invalid RR camera matrices"));
+            }
+            (distance, scalar_formats)
+        }
+    };
+
+    for (label, resource, formats) in [
+        ("RR HDR color", guides.color, rgba.as_slice()),
+        ("RR depth", guides.depth, &[DXGI_FORMAT_R32_FLOAT]),
+        ("RR motion", guides.motion, &motion_formats),
+        ("RR diffuse albedo", guides.diffuse_albedo, &albedo),
+        ("RR specular albedo", guides.specular_albedo, &albedo),
+        ("RR packed normals", guides.normals_roughness, &rgba),
+        ("RR reflections", reflections, &reflection_formats),
+    ]
+    .into_iter()
+    .chain(
+        guides
+            .depth_of_field
+            .map(|dof| ("RR depth of field guide", dof, &[DXGI_FORMAT_R16_FLOAT][..])),
+    ) {
+        // SAFETY: the descriptor is immutable and returned by value.
+        let desc = unsafe { resource.GetDesc() };
+        if !on_device(resource, device)
+            || resource.as_raw() == guides.output.as_raw()
+            || !texture_shape(&desc, INPUT, formats)
+            || desc
+                .Flags
+                .contains(D3D12_RESOURCE_FLAG_DENY_SHADER_RESOURCE)
+        {
+            return Err(Error::Invalid(label));
+        }
+    }
+
+    // SAFETY: the descriptor is immutable and returned by value.
+    let output = unsafe { guides.output.GetDesc() };
+    let mut support = D3D12_FEATURE_DATA_FORMAT_SUPPORT {

+        ..Default::default()
+    };

+            .contains(D3D12_RESOURCE_FLAG_ALLOW_UNORDERED_ACCESS)

+            .Support2
+            .contains(D3D12_FORMAT_SUPPORT2_UAV_TYPED_STORE)

+        ));
+    }

+fn on_device(child: &ID3D12DeviceChild, device: &ID3D12Device) -> bool {
+    let mut actual = None::<ID3D12Device>;
+    // SAFETY: the out pointer is valid and uses the D3D12 device IID.
+    unsafe { child.GetDevice(&mut actual) }.is_ok()
+        && actual
+            .as_ref()
+            .is_some_and(|actual| actual.as_raw() == device.as_raw())
+}

+fn texture_shape(desc: &D3D12_RESOURCE_DESC, size: [u32; 2], formats: &[DXGI_FORMAT]) -> bool {
+    desc.Dimension == D3D12_RESOURCE_DIMENSION_TEXTURE2D
+        && desc.Width == u64::from(size[0])
+        && desc.Height == size[1]
+        && desc.DepthOrArraySize == 1
+        && desc.SampleDesc.Count == 1
+        && formats.contains(&desc.Format)
+}

+fn validate_temporal(frame: &Frame) -> Result<(), Error> {
+    if frame.jitter.iter().any(|v| !v.is_finite())
+        || frame
+            .motion_scale
+            .iter()
+            .any(|v| !v.is_finite() || *v == 0.0)
+        || [frame.pre_exposure, frame.frame_time_ms]
+            .iter()
+            .any(|v| !v.is_finite() || *v <= 0.0)
+    {
+        return Err(Error::Invalid(
+            "RR frame timing, exposure, jitter, or motion scale",
+        ));
+    }
+    Ok(())

+fn invertible(matrix: &[f32; 16]) -> bool {
+    if matrix.iter().any(|v| !v.is_finite()) {
+        return false;
+    }
+    let mut rows = std::array::from_fn::<_, 4, _>(|row| {
+        std::array::from_fn::<_, 4, _>(|col| f64::from(matrix[row * 4 + col]))
+    });
+    for col in 0..4 {
+        let pivot = (col..4)
+            .max_by(|a, b| rows[*a][col].abs().total_cmp(&rows[*b][col].abs()))
+            .unwrap();
+        if rows[pivot][col] == 0.0 {
+            return false;
+        }
+        rows.swap(col, pivot);
+        let row = rows[col];
+        for next in &mut rows[col + 1..] {
+            let scale = next[col] / row[col];
+            for k in col + 1..4 {
+                next[k] -= scale * row[k];
+            }
+        }
+    }
+    true
+}

+fn allocate(
+    device: &ID3D12Device,
+    desc: &D3D12_RESOURCE_DESC,
+    heap: D3D12_HEAP_TYPE,
+    state: D3D12_RESOURCE_STATES,
+) -> ID3D12Resource {
+    let mut resource = None;
+    // SAFETY: descriptor and output pointer are live. The fixture holds each texture across the fence.
+    unsafe {
+        device
+            .CreateCommittedResource(
+                &D3D12_HEAP_PROPERTIES {
+                    Type: heap,
+                    CreationNodeMask: 1,
+                    VisibleNodeMask: 1,
+                    ..Default::default()
+                },
+                D3D12_HEAP_FLAG_NONE,
+                desc,
+                state,
+                None,
+                &mut resource,
+            )
+            .unwrap();
+    }
+    resource.unwrap()
+}

+fn transition(
+    list: &ID3D12GraphicsCommandList,
+    texture: &ID3D12Resource,
+    from: D3D12_RESOURCE_STATES,
+    to: D3D12_RESOURCE_STATES,
+) {
+    // SAFETY: this is a borrowed interface; the barrier's union does not release it.
+    let pResource = unsafe { std::mem::transmute_copy(texture) };
+    let barrier = D3D12_RESOURCE_BARRIER {
+        Type: D3D12_RESOURCE_BARRIER_TYPE_TRANSITION,
+        Anonymous: D3D12_RESOURCE_BARRIER_0 {
+            Transition: std::mem::ManuallyDrop::new(D3D12_RESOURCE_TRANSITION_BARRIER {
+                pResource,
+                Subresource: D3D12_RESOURCE_BARRIER_ALL_SUBRESOURCES,
+                StateBefore: from,
+                StateAfter: to,
+            }),
+        },
+        ..Default::default()
+    };
+    // SAFETY: fixture owns this open list and tracks each texture state.
+    unsafe { list.ResourceBarrier(&[barrier]) };
+}

+fn reset_provider() {
+    let mut provider = PROVIDER.lock().unwrap();
+    let _sdk = SDK.lock().unwrap();
+    if let Some(ngx) = provider.ngx.take() {

+        }
+    }
+    *provider = Provider::default();
