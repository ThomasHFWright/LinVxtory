+fn record(
+    list: &HostList,
+    backing: &crate::resource::HostTexture,
+    state: D3D12_RESOURCE_STATES,
+    kind: Kind,
+    ffx_output_serial: u64,
+    pre_exposure: Option<f32>,
+    begun: u64,
+) -> Option<()> {
+    if TAPS.fetch_update(Ordering::Relaxed, Ordering::Relaxed, |n| (n < MAX_TAPS).then_some(n + 1)).is_err() {
+        return None;
+    }
+    let (_, format, _) = kind.layout();
+    let buffer = crate::host::host()?.read_back_buffer(BYTES)?;
+    let texture = &backing.texture;
+    let copy = D3D12_RESOURCE_STATE_COPY_SOURCE;
+    // SAFETY: the exact tracked source is a single native subresource of
+    // the expected typed format. The four distinct footprints obey D3D12
+    // offset/row alignment and lie inside this fenced readback buffer.
+    unsafe {
+        list.list.barriers(&[transition(texture, state, copy)]);
+        let src = D3D12_TEXTURE_COPY_LOCATION {
+            pResource: std::mem::transmute_copy(texture),
+            Type: D3D12_TEXTURE_COPY_TYPE_SUBRESOURCE_INDEX,
+            Anonymous: D3D12_TEXTURE_COPY_LOCATION_0 { SubresourceIndex: 0 },
+        };
+        for (i, [left, top]) in SITES.into_iter().enumerate() {
+            let dst = D3D12_TEXTURE_COPY_LOCATION {
+                pResource: std::mem::transmute_copy(&buffer),
+                Type: D3D12_TEXTURE_COPY_TYPE_PLACED_FOOTPRINT,
+                Anonymous: D3D12_TEXTURE_COPY_LOCATION_0 {
+                    PlacedFootprint: D3D12_PLACED_SUBRESOURCE_FOOTPRINT {
+                        Offset: i as u64 * REGION_BYTES,
+                        Footprint: D3D12_SUBRESOURCE_FOOTPRINT {
+                            Format: format, Width: BOX, Height: BOX, Depth: 1, RowPitch: ROW_BYTES,
+                        },
+                    },
+                },
+            };
+            let region = D3D12_BOX { left, top, front: 0, right: left + BOX, bottom: top + BOX, back: 1 };
+            list.list.CopyTextureRegion(&dst, 0, 0, 0, &src, Some(&region));
+        }
+        list.list.barriers(&[transition(texture, copy, state)]);
+    }
+    list.keep(texture);
+    list.keep(&buffer);
+    list.add_lighting_tap(Tap {
+        kind, begun, source_serial: backing.serial,
+        ffx_output_serial, pre_exposure, buffer, accepted: None,
+    });
+    Some(())

+pub(super) fn probe(
+    state: &State, list: &HostList, pipeline: &HostPipeline, groups: [u32; 3], before: Option<u64>,
+) -> Option<u64> {
+    if groups != [160, 180, 1] || pipeline.stages.first()?.shader.hash != PROBE || (before.is_none() && !selected()) {
+        return None;
+    }
+    let (title, kind) = if before.is_some() {
+        let View::Unordered(title, 0, _) = probe_view(state, 176)? else { return None };
+        if !matches!(probe_view(state, 184), Some(View::Unordered(alias, 0, _)) if alias == title) {
+            return None;
+        }
+        (title, Kind::GiOutput)
+    } else {
+        let view = probe_view(state, 120)?;
+        let view = if matches!(view, View::Placed(false, _, _)) {
+            crate::descriptor::host::placed_texture(&view)?
+        } else {
+            view
+        };
+        let View::Shader(title, _) = view else { return None };
+        (title, Kind::GiInput)
+    };
+    let (_, format, _) = kind.layout();
+    let (resource, states) = rendered_at(list, title, format, VIEW)?;
+    let required = if before.is_some() { D3D12_RESOURCE_STATE_UNORDERED_ACCESS } else { D3D12_RESOURCE_STATE_NON_PIXEL_SHADER_RESOURCE };
+    if !states.contains(required) {
+        return None;
+    }
+    let begun = before.unwrap_or_else(crate::frames_begun);
+    record(list, &resource, states, kind, 0, None, begun)?;
+    Some(begun)
+}
