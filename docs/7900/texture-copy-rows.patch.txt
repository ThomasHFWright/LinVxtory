+//! CPU tile and detile throughput using the same public API on both revisions.
+
+use std::hint::black_box;
+use std::time::Instant;
+use victory_formats::texture::{Desc, Dimension, Layout, Swizzle};
+
+fn main() {
+    for (label, block, element_bytes, swizzle) in [
+        ("R8", 1, 1, Swizzle::SW_64KB_S),
+        ("RGBA8", 1, 4, Swizzle::SW_4KB_D),
+        ("BC1", 4, 8, Swizzle::SW_64KB_S),
+        ("BC7", 4, 16, Swizzle::SW_64KB_D),
+    ] {
+        let layout = Layout::new(&Desc {
+            dimension: Dimension::Tex2d, width: 1021, height: 767, depth: 1, mips: 1,
+            block, element_bytes, swizzle, samples: 1,
+        }).unwrap();
+        let mip = &layout.mips[0];
+        let row = mip.width as usize * element_bytes as usize;
+        let pitch = row + 3;
+        let mut tiled = vec![0; layout.size as usize];
+        let input: Vec<u8> = (0..pitch * mip.height as usize).map(|i| (i ^ (i >> 8)) as u8).collect();
+        let mut output = vec![0; input.len()];
+        for detile in [false, true] {
+            let mut times = [0.0; 7];
+            for time in &mut times {
+                let start = Instant::now();
+                for _ in 0..32 {
+                    if detile {
+                        layout.detile(black_box(&tiled), 0, 0, black_box(&mut output), pitch, 0).unwrap();
+                    } else {
+                        layout.tile(black_box(&mut tiled), 0, 0, black_box(&input), pitch, 0).unwrap();
+                    }
+                }
+                *time = start.elapsed().as_secs_f64();
+            }
+            times.sort_by(f64::total_cmp);
+            let mib = (32 * row * mip.height as usize) as f64 / 1048576.0;
+            let op = if detile { "detile" } else { "tile" };
+            println!("{label} {op} {:.1} MiB/s", mib / times[3]);
+        }
+        for (a, b) in input.chunks(pitch).zip(output.chunks(pitch)) {
+            assert_eq!(&a[..row], &b[..row]);
+        }
+    }
+}

+    /// Detects the uniform runs of mip `m`'s rows for copy widths 4, 8, 16 or 32.
+    fn new(layout: &Layout, m: &Mip, n: usize) -> Option<RowRuns> {
+        let pat = &layout.pattern[0];
+        let bw = layout.block[0] as usize;
+        let mut k = 1usize;
+        while k < pat.len() && pat[k] == pat[k - 1] + n as u32 {
+            k += 1;
+        }
+        let uniform = k.is_power_of_two()
+            && pat.len() == bw
+            && bw.is_multiple_of(k)
+            && (1..pat.len()).all(|i| (pat[i] == pat[i - 1] + n as u32) == !i.is_multiple_of(k));
+        let w = k * n;
+        // GFX10 tail shifts use whole micro tiles, which start on run boundaries.
+        let shift = m.tail.map_or(0, |t| t.x as usize);
+        if !uniform || !matches!(w, 4 | 8 | 16 | 32) || !shift.is_multiple_of(k) {
+            return None;
+        }

+        let column = |x| layout.column(m, x as u32) as usize;
+        let cols = (0..tail_start).step_by(k).map(column).collect();
