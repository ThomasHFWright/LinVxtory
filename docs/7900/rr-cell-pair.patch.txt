+//! Uncompressed scanline OpenEXR, preserving GPU binary16 channel bits.
+
+use std::io::{self, Write};
+
+pub(super) fn write(
+    mut out: impl Write,
+    [width, height]: [u32; 2],
+    rgba: &[u16],
+    pre_exposure: f32,
+) -> io::Result<()> {
+    assert_eq!(rgba.len(), width as usize * height as usize * 4);
+    let mut header = Vec::new();
+    header.extend(20_000_630u32.to_le_bytes());
+    header.extend(2u32.to_le_bytes());
+    let mut attr = |name: &[u8], kind: &[u8], bytes: &[u8]| {
+        header.extend(name);
+        header.push(0);
+        header.extend(kind);
+        header.push(0);
+        header.extend((bytes.len() as u32).to_le_bytes());
+        header.extend(bytes);
+    };
+    const CHANNELS: [(u8, usize); 4] = [(b'A', 3), (b'B', 2), (b'G', 1), (b'R', 0)];
+    let mut channels = Vec::new();
+    for (name, _) in CHANNELS {
+        channels.extend([name, 0]);
+        channels.extend(1u32.to_le_bytes()); // HALF
+        channels.extend([0; 4]);
+        channels.extend(1u32.to_le_bytes()); // xSampling
+        channels.extend(1u32.to_le_bytes()); // ySampling
+    }
+    channels.push(0);
+    let window: Vec<_> = [0, 0, width - 1, height - 1]
+        .into_iter()
+        .flat_map(u32::to_le_bytes)
+        .collect();
+    let center = [0u8; 8];
+    let one = 1.0f32.to_le_bytes();
+    attr(b"channels", b"chlist", &channels);
+    attr(b"compression", b"compression", &[0]);
+    attr(b"dataWindow", b"box2i", &window);
+    attr(b"displayWindow", b"box2i", &window);
+    attr(b"lineOrder", b"lineOrder", &[0]);
+    attr(b"pixelAspectRatio", b"float", &one);
+    attr(b"screenWindowCenter", b"v2f", &center);
+    attr(b"screenWindowWidth", b"float", &one);
+    attr(b"victoryPreExposure", b"float", &pre_exposure.to_le_bytes());
+    header.push(0);
+
+    let row_bytes = width * 8;
+    let first_scanline = header.len() as u64 + u64::from(height) * 8;
+    out.write_all(&header)?;
+    for y in 0..height {
+        let offset = first_scanline + u64::from(y) * u64::from(8 + row_bytes);
+        out.write_all(&offset.to_le_bytes())?;
+    }
+
+    let mut line = Vec::with_capacity((row_bytes + 8) as usize);
+    for (y, pixels) in rgba.chunks_exact(width as usize * 4).enumerate() {
+        line.clear();
+        line.extend((y as u32).to_le_bytes());
+        line.extend(row_bytes.to_le_bytes());
+        for (_, i) in CHANNELS {
+            for &word in pixels.iter().skip(i).step_by(4) {
+                line.extend(word.to_le_bytes());
+            }
+        }
+        out.write_all(&line)?;
+    }
+    Ok(())
+}
