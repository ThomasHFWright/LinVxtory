+//! Renderer-only capture commands, adapted from OctoAlgo's inkVI-dive
+//! (479a9c20ed5bc4843bdf8c6ad1a199f939f991cb, GPL-3.0).

+use std::io::{Read, Result};

+const MAX_REQUEST: u64 = 8;
+const MAX_TIMED: u64 = 32;
+static REQUESTS: Mutex<VecDeque<RangeInclusive<u64>>> = Mutex::new(VecDeque::new());

+fn control() -> Option<&'static Path> {
+    static FOLDER: OnceLock<Option<PathBuf>> = OnceLock::new();
+    FOLDER
+        .get_or_init(|| std::env::var_os("VICTORY_CAPTURE_CONTROL").map(PathBuf::from))
+        .as_deref()

+pub(crate) fn request_only() -> bool {
+    static ON: OnceLock<bool> = OnceLock::new();
+    *ON.get_or_init(|| std::env::var_os("VICTORY_CAPTURE_REQUEST_ONLY").is_some_and(|v| v == "1"))
+}

+pub(crate) fn tracks_world() -> bool {

+fn count(command: &str) -> Option<u64> {
+    let mut words = command.split_whitespace();
+    if words.next()? != "cap" {
+        return None;
+    }
+    let n = words.next().map_or(Some(1), |s| s.parse().ok())?;
+    ((1..=MAX_REQUEST).contains(&n) && words.next().is_none()).then_some(n)
+}
+
+fn range(present: u64, count: u64) -> Option<RangeInclusive<u64>> {
+    Some(present.checked_add(1)?..=present.checked_add(count)?)
+}
+
+fn read_command(path: &Path) -> Result<String> {
+    let mut command = String::new();
+    std::fs::File::open(path)?
+        .take(64)
+        .read_to_string(&mut command)?;
+    if command.len() == 64 {
+        command.clear();
+    }
+    Ok(command)
+}

+pub(crate) fn present(frame: u64, world: Option<u64>, enabled: bool) {
+    let Some(folder) = control() else { return };
+    let pending = folder.join("cmd");
+    let claimed = folder.join("cmd.processing");
+    if std::fs::rename(pending, &claimed).is_err() {
+        return;
+    }
+    let mut requests = REQUESTS.lock().unwrap_or_else(|error| error.into_inner());
+    // Retain recent requests for asynchronously submitted Presents. One request
+    // at a time keeps synchronous 4K readbacks bounded.
+    let command = read_command(&claimed).unwrap_or_default();
+    let capture_count = count(&command);
+    let reply = match capture_count {
+        None => "error: expected cap or cap N, where N is 1..8".to_owned(),
+        Some(_) if !enabled => "error: cap needs VICTORY_CAPTURE".to_owned(),
+        Some(_) if requests.back().is_some_and(|active| frame < *active.end()) => {
+            "error: a capture request is still active".to_owned()
+        }
+        Some(n) => match range(frame, n) {
+            Some(frames) => {
+                if requests.len() == 8 {
+                    requests.pop_front();
+                }
+                let reply = format!(
+                    "cap: presents {}..{} are saved",
+                    frames.start(),
+                    frames.end()
+                );
+                requests.push_back(frames);
+                reply
+            }
+            None => "error: capture frame range overflowed".to_owned(),
+        },
+    };
+    drop(requests);
+    tracing::info!(frame, "capture control: {reply}");
+    let world_offset = world
+        .and_then(|first| frame.checked_sub(first))
+        .map_or_else(|| "-".to_owned(), |n| n.to_string());
+    let stamp = format!("present {frame} world {world_offset}");
+    let description = capture_count.map_or_else(
+        || "invalid capture request".to_owned(),
+        |count| format!("cap {count}"),
+    );
+    let world = world.map_or_else(|| "unknown".to_owned(), |n| n.to_string());
+    let report = format!(
+        "[{stamp}] > {description}\n{reply}\nobserved_present={frame}\nworld_frame={world}\n"
+    );
+    let published = std::fs::write(folder.join("reply.txt"), report)
+        .and_then(|()| std::fs::write(folder.join("done"), format!("{stamp}\n")));
+    if let Err(error) = published {
+        tracing::warn!(%error, "capture control reply could not be written");
+    }
+    let _ = std::fs::remove_file(claimed);
+}

+fn seconds_window(value: &str) -> Option<(f64, f64)> {
+    let (a, b) = value.split_once('-')?;
+    let a = a.trim().parse::<f64>().ok()?;
+    let b = b.trim().parse::<f64>().ok()?;

+    static WINDOW: OnceLock<Option<(f64, f64)>> = OnceLock::new();

+pub(crate) fn in_seconds(frame: u64) -> bool {
+    let Some((from, to)) = window() else {
+        return false;
+    };
+    let Some(start) = WORLD_AT.get() else {
+        return false;
+    };
+    // This is queried in host Present order. Check wall time once per frame
+    // and count selected frames independently of the stock sampling cadence.
+    static STATE: Mutex<(u64, bool, u64)> = Mutex::new((0, false, 0));
+    let mut state = STATE.lock().unwrap_or_else(|error| error.into_inner());
+    if frame > state.0 {
+        let select = state.2 < MAX_TIMED && (from..=to).contains(&start.elapsed().as_secs_f64());
+        *state = (frame, select, state.2 + u64::from(select));
+    }
+    frame == state.0 && state.1
+}
