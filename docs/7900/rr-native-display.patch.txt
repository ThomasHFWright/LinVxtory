+pub(crate) fn queue_ingress<'a>(
+    queue: usize,
+    lists: impl Iterator<Item = &'a HostList>,
+) -> Option<QueueIngress> {
+    let target = target()?;
+    let frame = crate::queue::presented() + 1;
+    if !window(frame, target) {
+        return None;
+    }
+    let handoffs: Vec<_> = lists
+        .filter_map(|list| {
+            let handoff_order = list.rr_order_handoff.load(Ordering::Acquire);
+            (handoff_order != 0).then(|| (list.epoch(), handoff_order))
+        })
+        .take(9)
+        .collect();

+        return None;
+    }
+    Some(QueueIngress {

+        frame,
+        before: order(),
+        handoffs,
+    })
