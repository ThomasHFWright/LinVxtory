+fn parse_request(entry: &str) -> Option<Request> {
+    let mut fields = entry.split(',');
+    let (hash, kind, slot, dwords) = (
+        fields.next()?,
+        fields.next()?,
+        fields.next()?,
+        fields.next()?,
+    );
+    if fields.next().is_some() {
+        return None;
+    }
+    if hash.len() != 16 || !hash.bytes().all(|c| c.is_ascii_hexdigit()) {
+        return None;
+    }
+    let kind = match kind {
+        "srv" => ResourceType::Srv,
+        "uav" => ResourceType::Uav,
+        _ => return None,
+    };
+    let dwords = dwords.parse().ok()?;
+    if dwords % 8 != 0 {
+        return None;
+    }
+    Some(Request {
+        hash: hash.to_ascii_lowercase(),
+        kind,
+        slot: slot.parse().ok()?,
+        dwords,
+        attempts: AtomicU64::new(0),
+    })

+        return Err("UAV is not the tracked writable texture");

+        .ok_or("unexpected radiance grid backing or typed format")?;

+    if bytes == 0

+    // SAFETY: the texture is held in the list's state map and the buffer is
+    // kept through reset (the list can be submitted again). Restore exactly
+    // the state the binding established so subsequent title barriers agree.
+    unsafe {
+        if transition {
+            list.list
+                .ResourceBarrier(&[crate::host::transition(texture, before, copy_state)]);
+        }
+        if whole {
+            // D3D12 requires whole-subresource copies for depth-stencil.
+            crate::capture::copy_texture_region(
+                &list.list, texture, mip_host, &buffer, footprint, None,
+            );
+        } else {
+            for i in 0..GRID * GRID {

+                    top: y,
+                    front: 0,
+                    right: x + 2,
+                    bottom: y + 2,
+                    back: 1,
+                };

+                    Offset: u64::from(i) * SLOT,
+                    Footprint: D3D12_SUBRESOURCE_FOOTPRINT {
+                        Format: footprint.Footprint.Format,
+                        Width: 2,
+                        Height: 2,
+                        Depth: 1,
+                        RowPitch: ROW,
+                    },
+                };
+                crate::capture::copy_texture_region(
+                    &list.list,
+                    texture,
+                    mip_host,
+                    &buffer,
+                    to,
+                    Some(&region),
+                );
+            }
+        }
+        if transition {
+            list.list
+                .ResourceBarrier(&[crate::host::transition(texture, copy_state, before)]);
+        }
+    }
+    list.keep(&buffer);
+    let map_format = footprint.Footprint.Format.0;
+    let metadata = format!(
+        "resource={resource:#x}\tserial={}\tepoch={}\tsize={width}x{height}\tmip={mip}\tsubresource={mip_host}\tplane={plane}\tformat={}\tview={view_format}\tcopy_format={map_format}\tbytes={element}\tgrid={}x{}\tgrid_scale={}\tlist={:#x}",

+        list.epoch(),
+        desc.Format.0,
+        grid.width,
+        grid.height,
+        grid.scale,
+        list.list.as_raw() as usize,
+    );

+        name: name.to_owned(),
+        metadata,
+        buffer,
+        footprint,
+        grid,
+        bytes: bytes as usize,
+        element: element as usize,
+        whole,
+    });

+    metadata: String,
+    buffer: ID3D12Resource,

+impl Readback {
+    pub(crate) fn save(&self) {
+        let Some(folder) = crate::capture::folder() else {
+            return;
+        };
+        let mut data = std::ptr::null_mut();
+        let range = D3D12_RANGE {
+            Begin: 0,
+            End: self.bytes,
+        };
+        // SAFETY: host calls this only after the GPU submission's wait.
+        let saved = unsafe {
+            match self.buffer.Map(0, Some(&range), Some(&mut data)) {
+                Ok(()) if !data.is_null() => {
+                    let bytes = std::slice::from_raw_parts(data.cast::<u8>(), self.bytes);
+                    let result = std::fs::File::create(folder.join(format!("{}.tsv", self.name)))
+                        .and_then(|file| self.write(BufWriter::new(file), bytes));
+                    self.buffer
+                        .Unmap(0, Some(&D3D12_RANGE { Begin: 0, End: 0 }));
+                    result
+                }
+                _ => Err(std::io::Error::other("readback map failed")),
+            }
+        };
+        if let Err(error) = saved {
+            tracing::warn!(name = self.name, %error, "RR guide sample was not saved");
+        }
+    }
+
+    fn write(&self, mut out: impl Write, bytes: &[u8]) -> std::io::Result<()> {
+        writeln!(out, "# {}", self.metadata)?;
+        writeln!(out, "x\ty\thex")?;
+        for i in 0..GRID * GRID {
+            let (x, y) = self.grid.position(i);
+            for row in 0..2 {
+                for col in 0..2 {
+                    let start = if self.whole {
+                        self.footprint.Offset as usize
+                            + (y + row) as usize * self.footprint.Footprint.RowPitch as usize
+                            + (x + col) as usize * self.element
+                    } else {
+                        i as usize * SLOT as usize
+                            + row as usize * ROW as usize
+                            + col as usize * self.element
+                    };
+                    let pixel = bytes
+                        .get(start..start + self.element)
+                        .ok_or_else(|| std::io::Error::other("texel outside readback"))?;
+                    write!(out, "{}\t{}\t", x + col, y + row)?;
+                    for byte in pixel {
+                        write!(out, "{byte:02x}")?;
+                    }
+                    writeln!(out)?;
+                }
+            }
+        }
+        out.flush()
+    }

+fn snapshot<const N: usize>(base: u64, size: u32, offset: u32) -> Option<[u8; N]> {
+    let end = offset.checked_add(u32::try_from(N).ok()?)?;
+    if end > size {
+        return None;
+    }
+    base.checked_add(u64::from(end))?;
+    let address = (base + u64::from(offset)) as *const u8;
+    let mut first = [0; N];
+    let mut second = [0; N];
+    (crate::host::read_into(address, &mut first)
+        && crate::host::read_into(address, &mut second)
+        && first == second)
+        .then_some(first)
