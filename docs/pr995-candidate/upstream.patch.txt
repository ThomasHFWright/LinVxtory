+pub(super) struct Semantics(pub(super) BTreeMap<(String, u32), u8>);
+
+impl Semantics {
+    pub(super) fn reads(dxil: &[u8]) -> Option<Self> {
+        let shader = Container::parse(dxil).ok()?;
+        let data = shader.part(FourCC::ISG1)?.data;
+        let base = crate::dxbc::u32_at(data, 4).ok()? as usize;
+        let mut reads = BTreeMap::new();
+        for (i, element) in signature(data).ok()?.iter().enumerate() {
+            let mask = *data.get(base + i * 32 + 25)?;
+            if element.kind == 0 && mask != 0 {
+                *reads
+                    .entry((element.name.to_ascii_uppercase(), element.index))
+                    .or_default() |= mask;
+            }
+        }
+        Some(Self(reads))
+    }
+
+    pub(super) fn prune(&self, elements: &mut Vec<SigElement>) {
+        elements.retain(|e| {
+            e.kind != 0 || self.0.contains_key(&(e.name.to_ascii_uppercase(), e.index))
+        });
+    }

+/// Measures the emitted declaration, including any reduction to lane counts.
+pub fn mesh_output_bytes(dxil: &[u8]) -> Option<u64> {
+    let shader = Container::parse(dxil).ok()?;
+    let psv = shader.part(FourCC::PSV0)?.data;
+    if crate::dxbc::u32_at(psv, 0).ok()? < 48 || *psv.get(28)? != 13 {
+        return None;
+    }
+    let count = |off| {
+        Some(u32::from(u16::from_le_bytes(
+            psv.get(off..off + 2)?.try_into().ok()?,
+        )))
+    };
+    let mut vertex = signature(shader.part(FourCC::OSG1)?.data).ok()?;
+    let mut primitive = match shader.part(FourCC::PSG1) {
+        Some(p) => signature(p.data).ok()?,
+        None => Vec::new(),
+    };
+    // Unallocated system values consume no output storage.
+    vertex.retain(|e| e.row.is_some());
+    primitive.retain(|e| e.row.is_some());
+    Some(container::mesh_output_bytes(
+        &vertex,
+        &primitive,
+        count(16)?,
+        count(18)?,
+    ))

+/// Require every surviving arbitrary pixel input to have its producer at the
+/// same location and with the same type. Leave incompatible pairs standalone.
+pub(super) fn matching_outputs(mesh: &[u8], pixel: &[u8]) -> Option<bool> {
+    let ms = Container::parse(mesh).ok()?;
+    let ps = Container::parse(pixel).ok()?;
+    let mut outputs = signature(ms.part(FourCC::OSG1)?.data).ok()?;
+    if let Some(prims) = ms.part(FourCC::PSG1) {
+        outputs.extend(signature(prims.data).ok()?);
+    }
+    let inputs = signature(ps.part(FourCC::ISG1)?.data).ok()?;
+    Some(inputs.iter().filter(|i| i.kind == 0).all(|i| {
+        outputs.iter().any(|o| {
+            o.kind == 0
+                && o.index == i.index
+                && o.name.eq_ignore_ascii_case(&i.name)
+                && o.comp == i.comp
+                && o.row == i.row
+                && o.col == i.col
+                && o.cols == i.cols
+        })
+    }))
+}

+/// Re-emit both stages using the standalone pixel shader's actual input reads.
+pub fn translate_linked(
+    mesh: &[u8],
+    pixel: &[u8],
+    standalone_pixel: &HostShader,
+) -> Option<(HostShader, HostShader, u64)> {
+    let reads = Semantics::reads(&standalone_pixel.dxil)?;
+    let ms = translate_selected(mesh, Some(&reads)).ok()?;
+    let bytes = mesh_output_bytes(&ms.dxil)?;
+    if bytes > MESH_OUTPUT_LIMIT {
+        return None;
+    }
+    let ps = translate_selected(pixel, Some(&reads)).ok()?;

+        || ps.stage != HwStage::Ps
+        || Semantics::reads(&ps.dxil)?.0 != reads.0
+        || !matching_outputs(&ms.dxil, &ps.dxil)?
+    {
+        return None;
+    }
+    Some((ms, ps, bytes))
