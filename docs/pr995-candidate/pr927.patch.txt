+pub(crate) fn half_view(view: &View) -> Option<View> {
+    // Buffer views: dimension 1 in the title's description of either kind.
+    const BUFFER: u32 = 1;
+    match view {

+            let mut desc = *desc;

+            Some(View::Shader(*resource, Some(desc)))
+        }

+//! Fresh descriptor table bounds and identity, without a host device.
+
+use std::hash::{Hash, Hasher};
+
+/// Half views can lie beyond a declared range's first descriptor. Keep
+/// them in both the allocation span and the caller's readable selections.
+pub(crate) fn span(
+    slot: u8,
+    offsets: impl Iterator<Item = u32>,
+    selected: &mut Vec<u32>,
+    half: &[(u8, u32)],
+) -> u32 {
+    selected.extend(
+        half.iter()
+            .filter(|&&(s, _)| s == slot)
+            .map(|&(_, at)| at / 8),
+    );
+    selected.sort_unstable();
+    selected.dedup();
+    offsets
+        .map(|at| at / 8)
+        .chain(selected.last().copied())
+        .max()
+        .unwrap_or(0)
+        + 1
+}
+
+/// Snapshot each descriptor's record and installed resident mip range.
+/// The half-view selection also changes the installed descriptor format.
+pub(crate) fn key(
+    table: u64,
+    slot: u8,
+    count: u32,
+    half: &[(u8, u32)],
+    mut entry: impl FnMut(u32) -> ([u32; 3], u32),
+) -> (u64, u32, u64) {
+    let mut hash = std::collections::hash_map::DefaultHasher::new();
+    for index in 0..count {
+        let (record, first_shown) = entry(index);
+        record.hash(&mut hash);
+        first_shown.hash(&mut hash);
+        half.contains(&(slot, index * 8)).hash(&mut hash);
+    }
+    (table, count, hash.finish())
+}

+//! Typed buffer loads that move 16-bit halves without the d16 form: a
+//! format load of whole dwords whose result the program reads only one
+//! half of, as a half float, moves 16-bit integers, like a d16 access
+//! ([`super::super::HostShader::half_views`]).
+
+use std::collections::HashSet;
+
+use super::{Fields, Inst, Lower, Operand, Reg, Role, RootType, Source, Value};
+
+impl<'a> Lower<'a> {
+    /// Whether every use of `value`, a typed buffer load's result, reads
+    /// one 16-bit half of it as a half float: a `v_fma_mix*` source with
+    /// its `op_sel_hi` bit set, or `v_cvt_f32_f16`. A program that takes a
+    /// format load apart like that reads a 16-bit integer view on the
+    /// console (the load gives the zero-extended integer, whose bits are a
+    /// half; a float view's value it would not take apart). A use is a
+    /// read as an operand ([`crate::ir::operand_reads`]): a later write of
+    /// the register reads the value only to keep it in the lanes EXEC
+    /// leaves alone. Follow that write and phis to their later uses. False
+    /// without a use.
+    pub(super) fn read_only_as_halves(&self, value: Value) -> bool {
+        let p = self.x.p;
+        let mut used = false;
+        let mut seen = HashSet::from([value]);
+        let mut todo = vec![value];
+        while let Some(value) = todo.pop() {
+            for b in &p.blocks {
+                for phi in b.phis.iter().filter(|phi| phi.args.contains(&value)) {
+                    if seen.insert(phi.value) {
+                        todo.push(phi.value);
+                    }
+                }
+                for i in &b.insts {
+                    if !i.reads.iter().any(|r| r.1 == value) {
+                        continue;
+                    }
+                    // A write may preserve inactive lanes or untouched bits, even
+                    // when the same register is also an explicit half operand.
+                    // Without a proof of complete overwrite, follow its new value.
+                    for &(reg, next) in &i.writes {
+                        if matches!(reg, Reg::V(_))
+                            && i.read(reg) == Some(value)
+                            && seen.insert(next)
+                        {
+                            todo.push(next);
+                        }
+                    }
+                    let operands = crate::ir::operand_reads(p, i);
+                    let regs: Vec<Reg> = i
+                        .reads
+                        .iter()
+                        .filter(|r| r.1 == value && operands.contains(&r.0))
+                        .map(|r| r.0)
+                        .collect();
+                    if regs.is_empty() {
+                        continue;
+                    }
+                    used = true;
+                    let Fields::Valu(v) = &i.fields else {
+                        return false;
+                    };
+                    match i.name() {
+                        "v_cvt_f32_f16" => {}
+                        "v_fma_mix_f32" | "v_fma_mixlo_f16" | "v_fma_mixhi_f16" => {
+                            let whole = v.src.iter().enumerate().any(|(k, op)| {
+                                matches!(*op, Operand::Vector { reg, .. } if regs.contains(&Reg::V(reg)))
+                                    && v.opsel_hi >> k & 1 == 0
+                            });
+                            if whole {
+                                return false;
+                            }
+                        }
+                        _ => return false,
+                    }
+                }
+            }
+        }
+        used
+    }
+
+    /// Whether `inst` is a format load of whole dwords (not d16) whose
+    /// every channel is read only as halves ([`Self::read_only_as_halves`]).
+    pub(super) fn format_load_read_as_halves(&self, inst: &Inst) -> bool {
+        let Fields::Buffer {
+            data: Some(Operand::Vector { reg, .. }),
+            ..
+        } = inst.fields
+        else {
+            return false;
+        };
+        let name = inst.name();
+        let count = match name
+            .strip_prefix("buffer_load_format_")
+            .or(name.strip_prefix("tbuffer_load_format_"))
+        {
+            Some("x") => 1,
+            Some("xy") => 2,
+            Some("xyz") => 3,
+            Some("xyzw") => 4,
+            _ => return false,
+        };
+        (0..count).all(|k| {
+            inst.write(Reg::V(reg + k))
+                .is_some_and(|v| self.read_only_as_halves(v))
+        })
+    }
+
+    /// The table places ([`Self::table_place`]) every access through which
+    /// moves halves: a d16 access, or a format load read only as halves
+    /// ([`Self::format_load_read_as_halves`]). A place the program reaches
+    /// otherwise too is left out (a format load of numbers beside one of
+    /// halves, a raw or scalar load, a descriptor chosen by control flow),
+    /// and so is every place of a table the program also picks from at run
+    /// time: what the program makes of such a view is not halves alone.
+    pub(super) fn places_read_as_halves(&self) -> std::collections::BTreeSet<(u8, u32)> {
+        let mut halves = std::collections::BTreeSet::new();
+        let mut other = std::collections::BTreeSet::new();
+        let mut picked = HashSet::new();
+        for d in &self.x.res.descriptors {
+            if d.role == Role::Sampler {
+                continue;
+            }
+            let inst = self
+                .x
+                .p
+                .blocks
+                .get(d.site.0)
+                .and_then(|b| b.insts.get(d.site.1));
+            let half = matches!(d.source, Source::Loaded { .. })
+                && inst.is_some_and(|i| {
+                    (i.name().starts_with("buffer_") && i.name().contains("_format_d16_"))
+                        || self.format_load_read_as_halves(i)
+                });
+            for leaf in d.source.leaves() {
+                let Source::Loaded { load, .. } = leaf else {
+                    continue;
+                };
+                let Some(a) = self.scalar.get(load) else {
+                    continue;
+                };
+                if a.dynamic.is_some() {
+                    if let Some(rp) = self.x.native.layout.get(a.param)
+                        && rp.root == RootType::Table
+                    {
+                        picked.insert(rp.slot);
+                    }
+                } else if let Some(place) = self.table_place(leaf) {
+                    if half {
+                        halves.insert(place);
+                    } else {
+                        other.insert(place);
+                    }
+                }
+            }
+        }
+        halves.retain(|p| !other.contains(p) && !picked.contains(&p.0));
+        halves
+    }

+/// A program's constant table starts after its instructions, unless a
+/// load at a fixed offset reads below that.
+#[test]
+fn the_code_table_starts_after_the_instructions() {
+    // s_nop 0, s_endpgm, then 1,000 dwords of constants.
+    let mut words = vec![0xBF80_0000, S_ENDPGM];
+    words.extend(0..1000u32);
+    let code: Vec<u8> = words.iter().flat_map(|w| w.to_le_bytes()).collect();
+    let p = ir::build(&code, false, 256).unwrap();
+    let at = |offset: u32, dynamic: bool| resource::Address {
+        param: resource::CODE,
+        offset,
+        dynamic: dynamic.then_some(ir::Value(0)),
+    };
+    let (computed, fixed_after, fixed_inside) = (at(0, true), at(12, false), at(4, false));
+    // A load at a computed offset from the code's start: the table is
+    // the constants only, not the whole program.
+    assert_eq!(code_table_start(&p, &[&computed], 0), 8);
+    assert_eq!(code_table_start(&p, &[&computed, &fixed_after], 0), 8);
+    // A load at a fixed offset inside the instructions keeps the start.
+    assert_eq!(code_table_start(&p, &[&computed, &fixed_inside], 0), 0);
+    // Loads past the instructions only: from the lowest of them.
+    assert_eq!(code_table_start(&p, &[&fixed_after], 12), 12);
+}

+/// [`lower_with`] in wave64 if `wave64`.
+fn lower_wave(
+    words: &[u32],
+    layout: Vec<RootParam>,
+    used: Vec<UsedResource>,
+    wave64: bool,
+) -> Result<Lowered, crate::ShaderError> {
+    let code: Vec<u8> = words.iter().flat_map(|w| w.to_le_bytes()).collect();
+    let sgprs: u32 = layout.iter().map(|p| p.sgprs).sum();

+        stage: HwStage::Cs,
+        wave64,
+        vgprs: 256,
+        layout,
+        used,
+        io: StageIo::Cs(ComputeIo {
+            // The layout's user SGPRs (at least two), 512 bytes of LDS.
+            rsrc2: 1 << 15 | sgprs.max(2) << 1,
+            threads: [1, 1, 1],
+        }),
+        code,
+    };
+    let p = ir::build(&native.code, wave64, 256).unwrap();
+    let st = structure::structure(&p).unwrap();

+    let io = io::raise(&p, &native).unwrap();
+    lower(&Parts {
+        native: &native,
+        p: &p,
+        st: &st,
+        io: &io,
+        res: &res,
+        inputs: vec![],
+        outputs: vec![],
+        prims: vec![],

+/// A half's value (sign, 5-bit exponent biased 15, 10-bit mantissa), as
+/// `legacyF16ToF32` widens it.
+fn half(h: u32) -> f32 {
+    let (s, e, m) = ((h >> 15) & 1, (h >> 10) & 31, h & 1023);
+    let v = match e {
+        0 => m as f32 * 2f32.powi(-24),
+        31 if m == 0 => f32::INFINITY,
+        31 => f32::NAN,
+        _ => (1024 + m) as f32 * 2f32.powi(e as i32 - 25),
+    };
+    if s == 1 { -v } else { v }
+}

+    const fn mubuf_idxen(op: u32, vdata: u8, vaddr: u8, srsrc: u8) -> [u32; 2] {
+        [
+            0x38 << 26 | op << 18 | 1 << 13,
+            0x80 << 24 | ((srsrc as u32 >> 2) & 0x1F) << 16 | (vdata as u32) << 8 | vaddr as u32,
+        ]
+    }
+    const LOAD_D16_XYZW: u32 = 0x83;
+    const STORE_D16_XYZW: u32 = 0x87;
+    const THREADS: u32 = 32;
+    // The table's second descriptor, 0x20 bytes on.
+    let mut second = s_load_dwordx4(12, 4);
+    second[1] |= 0x20;
+    let mut words = vec![s_mov_b32(5, 0x80)];
+    words.extend_from_slice(&s_load_dwordx4(8, 4));
+    words.extend_from_slice(&second);
+    // v[1:2] = first[thread] (four halves), second[thread] = v[1:2].
+    words.extend_from_slice(&mubuf_idxen(LOAD_D16_XYZW, 1, 0, 8));
+    words.extend_from_slice(&mubuf_idxen(STORE_D16_XYZW, 1, 0, 12));
+    words.push(S_ENDPGM);
+    let view = |kind: ResourceType, table_offset_dwords: u32| UsedResource {
+        kind,
+        root: RootType::Table,
+        table_offset_dwords,
+        lower_bound: table_offset_dwords / 8,
+        stride: 4,
+        dwords: 4,
+        user_data: 4,
+        slot: 1,
+    };
+    let native = NativeProgram {
+        stage: HwStage::Cs,
+        wave64: false,
+        vgprs: 8,
+        layout: vec![
+            RootParam {
+                root: RootType::Constants,
+                table_offset_dwords: 0,
+                sgprs: 4,
+                base_register: 0,
+                user_data: 0,
+                slot: 0,
+            },
+            RootParam {
+                root: RootType::Table,
+                table_offset_dwords: 0,
+                sgprs: 1,
+                base_register: 0,
+                user_data: 4,
+                slot: 1,
+            },
+        ],
+        used: vec![view(ResourceType::Srv, 0), view(ResourceType::Uav, 8)],
+        io: StageIo::Cs(ComputeIo {
+            rsrc2: 5 << 1,
+            threads: [THREADS, 1, 1],
+        }),
+        code: words.iter().flat_map(|w| w.to_le_bytes()).collect(),
+    };
+    let p = ir::build(&native.code, native.wave64, native.vgprs).expect("ir::build");
+    let st = structure::structure(&p).expect("structure");
+    let res =
+        resource::track(&p, &native.layout, native.stage, Some(&st)).expect("resource::track");
+    let io = io::raise(&p, &native).expect("io::raise");
+    let lowered = lower::lower(&lower::Parts {
+        native: &native,
+        p: &p,
+        st: &st,
+        io: &io,
+        res: &res,
+        inputs: Vec::new(),
+        outputs: Vec::new(),
+        prims: Vec::new(),
+        mesh_stride: 0,
+    })
+    .expect("lower");
+    assert_eq!(lowered.half_views, [(1, 0), (1, 8)]);
+    let shader = HostShader {
+        stage: HwStage::Cs,
+        dxil: lowered.dxil,
+        user_data: lowered.user_data,
+        samplers: lowered.samplers,
+        static_samplers: Vec::new(),
+        root_cbvs: lowered.root_cbvs,
+        root_srvs: lowered.root_srvs,
+        root_uavs: lowered.root_uavs,
+        root_indexed_views: res.root_indexed_views,
+        half_views: lowered.half_views,
+    };
+    if let Ok(mut v) = validator::Validator::load(validator::Validator::default_path().as_deref()) {
+        let mut dxil = shader.dxil.clone();
+        v.validate(&mut dxil).expect("DXIL validation failed");
+    }
+    // A raw store through the same table: no halves, nothing named.
+    assert!(build_valu_cs(&[]).half_views.is_empty());
+
+    // Four halves a thread: numbers, both zeros, subnormals, infinities,
+    // not-a-numbers of both signs, and integers past the largest number's
+    // pattern, as indices are.
+    const SPECIAL: [u16; 16] = [
+        0x0000, 0x0001, 0x03ff, 0x0400, 0x3c00, 0x7bff, 0x7c00, 0x7c01, 0x7e00, 0x7fff, 0x8000,
+        0x8001, 0xbc00, 0xfc00, 0xfc01, 0xffff,
+    ];
+    let halves: Vec<u16> = (0..4 * THREADS as usize)
+        .map(|i| match i {
+            0..16 => SPECIAL[i],
+            _ => (i as u16).wrapping_mul(0x0203).wrapping_add(0x7b00),
+        })
+        .collect();
+    let data: Vec<u32> = halves
+        .chunks(2)
+        .map(|pair| u32::from(pair[0]) | u32::from(pair[1]) << 16)
+        .collect();
+    let _one = ONE_AT_A_TIME.lock().unwrap_or_else(|e| e.into_inner());
+    let mut ran = 0;
+    for (label, warp) in [("hardware", false), ("warp", true)] {
+        let Ok(host) = Host::new(warp) else {
+            eprintln!("no SM 6.6 {label} device; skipped");
+            continue;
+        };
+        let (root, pso) = host
+            .compute_pipeline(&shader)
+            .unwrap_or_else(|e| panic!("{label}: compute_pipeline failed: {e}"));
+        let mut ud = vec![0u32; shader.user_data_dwords().max(1) as usize];
+        for u in &shader.user_data {
+            ud[u.index as usize] = match (u.root, u.slot) {
+                (RootType::Table, 1) => 0,
+                other => panic!("unexpected user data {other:?}"),
+            };
+        }
+        let out = host
+            .dispatch(
+                &root,
+                &pso,
+                &Dispatch {
+                    roots: vec![(0, Root::Constants(ud))],
+                    views: &[
+                        View::Buffer {
+                            data: &data,
+                            stride: Stride::Format(DXGI_FORMAT_R16G16B16A16_UINT, 8),
+                        },
+                        View::HalfUintUav(THREADS),
+                    ],
+                    samplers: &[],
+                    groups: [1, 1, 1],
+                },
+            )
+            .unwrap_or_else(|e| panic!("{label}: dispatch failed: {e}"));
+        assert_eq!(
+            out[0], data,
+            "{label}: the halves stored are the halves loaded"
+        );
+        ran += 1;
+    }
+    assert!(ran > 0, "no SM 6.6 device available");
+}

+/// The places a program that moves halves names are the descriptors it
+/// loads, not the start of the range their used-resource entry describes:
+/// with one entry for a range from dword 0, a load at byte 0x20 names
+/// dword 8 (a host view there in a 16-bit float format is the one shown
+/// as its unsigned twin). A descriptor picked at run time (the load's
+/// offset in an SGPR) names nothing and keeps the float path: which slot
+/// it reads, and in which format, is known only then.
+#[test]
+fn half_places_are_the_descriptors_the_program_loads() {
+    /// MUBUF `op vdata, vaddr, s[srsrc:srsrc+3], 0 idxen`.
+    const fn mubuf_idxen(op: u32, vdata: u8, vaddr: u8, srsrc: u8) -> [u32; 2] {
+        [
+            0x38 << 26 | op << 18 | 1 << 13,
+            0x80 << 24 | ((srsrc as u32 >> 2) & 0x1F) << 16 | (vdata as u32) << 8 | vaddr as u32,
+        ]
+    }
+    const LOAD_D16_XYZW: u32 = 0x83;
+    const STORE_D16_XYZW: u32 = 0x87;
+    let lowered = |words: &[u32]| {
+        let range = |kind: ResourceType, table_offset_dwords: u32| UsedResource {
+            kind,
+            root: RootType::Table,
+            table_offset_dwords,
+            lower_bound: table_offset_dwords / 8,
+            stride: 4,
+            dwords: 4,
+            user_data: 4,
+            slot: 1,
+        };
+        let native = NativeProgram {
+            stage: HwStage::Cs,
+            wave64: false,
+            vgprs: 8,
+            layout: vec![
+                RootParam {
+                    root: RootType::Constants,
+                    table_offset_dwords: 0,
+                    sgprs: 4,
+                    base_register: 0,
+                    user_data: 0,
+                    slot: 0,
+                },
+                RootParam {
+                    root: RootType::Table,
+                    table_offset_dwords: 0,
+                    sgprs: 1,
+                    base_register: 0,
+                    user_data: 4,
+                    slot: 1,
+                },
+            ],
+            // One entry for a range of views from dword 0, one at 16.
+            used: vec![range(ResourceType::Srv, 0), range(ResourceType::Uav, 16)],
+            io: StageIo::Cs(ComputeIo {
+                rsrc2: 5 << 1,
+                threads: [32, 1, 1],
+            }),
+            code: words.iter().flat_map(|w| w.to_le_bytes()).collect(),
+        };
+        let p = ir::build(&native.code, native.wave64, native.vgprs).expect("ir::build");
+        let st = structure::structure(&p).expect("structure");
+        let res =
+            resource::track(&p, &native.layout, native.stage, Some(&st)).expect("resource::track");
+        let io = io::raise(&p, &native).expect("io::raise");
+        lower::lower(&lower::Parts {
+            native: &native,
+            p: &p,
+            st: &st,
+            io: &io,
+            res: &res,
+            inputs: Vec::new(),
+            outputs: Vec::new(),
+            prims: Vec::new(),
+            mesh_stride: 0,
+        })
+        .expect("lower")
+    };
+    // `s_load_dwordx4 s[dst:dst+3], s[4:5], offset` (soffset null), or with
+    // the offset in `soffset` instead.
+    let at = |dst: u8, offset: u32| {
+        let mut w = s_load_dwordx4(dst, 4);
+        w[1] |= offset;
+        w
+    };
+    let picked = |dst: u8, soffset: u8| {
+        let mut w = s_load_dwordx4(dst, 4);
+        w[1] = u32::from(soffset) << 25;
+        w
+    };
+    let mut fixed = vec![s_mov_b32(5, 0x80)];
+    fixed.extend_from_slice(&at(8, 0x20));
+    fixed.extend_from_slice(&at(12, 0x40));
+    fixed.extend_from_slice(&mubuf_idxen(LOAD_D16_XYZW, 1, 0, 8));
+    fixed.extend_from_slice(&mubuf_idxen(STORE_D16_XYZW, 1, 0, 12));
+    fixed.push(S_ENDPGM);
+    assert_eq!(lowered(&fixed).half_views, [(1, 8), (1, 16)]);
+    // The load's offset in s20, from the first lane's thread index
+    // (`v_readfirstlane_b32 s20, v0`): picked at run time.
+    let mut runtime = vec![s_mov_b32(5, 0x80), vop1(0x02, 20, vgpr(0))];
+    runtime.extend_from_slice(&picked(8, 20));
+    runtime.extend_from_slice(&at(12, 0x40));
+    runtime.extend_from_slice(&mubuf_idxen(LOAD_D16_XYZW, 1, 0, 8));
+    runtime.extend_from_slice(&mubuf_idxen(STORE_D16_XYZW, 1, 0, 12));
+    runtime.push(S_ENDPGM);
+    // The dynamic source can alias the fixed destination in the same table.
+    // Both keep legacy float conversion until their ranges can be proved disjoint.
+    assert!(lowered(&runtime).half_views.is_empty());
+}

+/// A format load of whole dwords (`buffer_load_format_x`, not d16) whose
+/// result the program only reads one half of as a half float
+/// (`v_cvt_f32_f16`) is moving 16-bit integers: the head program reads the
+/// face network's output so, from an `R16_UINT` view. It names its view for
+/// the host and loads the integer: 0x3c00 comes through as 1.0. Used as a
+/// whole float instead (`v_add_f32`), the load stays a float load and names
+/// nothing; so does a load of halves beside a load of the same view read as
+/// a whole float (a view of numbers the program also picks one half float
+/// out of, as hundreds of the title's pixel programs do).
+#[test]
+fn a_format_load_read_as_halves_loads_the_integer() {
+    use super::host::Stride;
+    use windows::Win32::Graphics::Dxgi::Common::DXGI_FORMAT_R16_UINT;
+    /// MUBUF `op vdata, vaddr, s[srsrc:srsrc+3], 0 idxen`.
+    const fn mubuf_idxen(op: u32, vdata: u8, vaddr: u8, srsrc: u8) -> [u32; 2] {
+        [
+            0x38 << 26 | op << 18 | 1 << 13,
+            0x80 << 24 | ((srsrc as u32 >> 2) & 0x1F) << 16 | (vdata as u32) << 8 | vaddr as u32,
+        ]
+    }
+    const LOAD_FORMAT_X: u32 = 0x00;
+    const V_CVT_F32_F16: u16 = 0x0B;
+    const V_ADD_F32: u8 = 0x03;
+    let lowered = |body: &[u32]| {
+        let view = |kind: ResourceType, table_offset_dwords: u32, stride: u32| UsedResource {
+            kind,
+            root: RootType::Table,
+            table_offset_dwords,
+            lower_bound: table_offset_dwords / 8,
+            stride,
+            dwords: 4,
+            user_data: 4,
+            slot: 1,
+        };
+        let mut second = s_load_dwordx4(12, 4);
+        second[1] |= 0x20;
+        let mut words = vec![s_mov_b32(5, 0x80)];
+        words.extend_from_slice(&s_load_dwordx4(8, 4));
+        words.extend_from_slice(&second);
+        // `body` leaves v2, stored raw.
+        words.extend_from_slice(body);
+        words.extend_from_slice(&buffer_store_dword(2, 12, 0));
+        words.push(S_ENDPGM);
+        let native = NativeProgram {
+            stage: HwStage::Cs,
+            wave64: false,
+            vgprs: 8,
+            layout: vec![
+                RootParam {
+                    root: RootType::Constants,
+                    table_offset_dwords: 0,
+                    sgprs: 4,
+                    base_register: 0,
+                    user_data: 0,
+                    slot: 0,
+                },
+                RootParam {
+                    root: RootType::Table,
+                    table_offset_dwords: 0,
+                    sgprs: 1,
+                    base_register: 0,
+                    user_data: 4,
+                    slot: 1,
+                },
+            ],
+            used: vec![view(ResourceType::Srv, 0, 4), view(ResourceType::Uav, 8, 0)],
+            io: StageIo::Cs(ComputeIo {
+                rsrc2: 5 << 1,
+                threads: [1, 1, 1],
+            }),
+            code: words.iter().flat_map(|w| w.to_le_bytes()).collect(),
+        };
+        let p = ir::build(&native.code, native.wave64, native.vgprs).expect("ir::build");
+        let st = structure::structure(&p).expect("structure");
+        let res =
+            resource::track(&p, &native.layout, native.stage, Some(&st)).expect("resource::track");
+        let io = io::raise(&p, &native).expect("io::raise");
+        let l = lower::lower(&lower::Parts {
+            native: &native,
+            p: &p,
+            st: &st,
+            io: &io,
+            res: &res,
+            inputs: Vec::new(),
+            outputs: Vec::new(),
+            prims: Vec::new(),
+            mesh_stride: 0,
+        })
+        .expect("lower");
+        HostShader {
+            stage: HwStage::Cs,
+            dxil: l.dxil,
+            user_data: l.user_data,
+            samplers: l.samplers,
+            static_samplers: Vec::new(),
+            root_cbvs: l.root_cbvs,
+            root_srvs: l.root_srvs,
+            root_uavs: l.root_uavs,
+            root_indexed_views: res.root_indexed_views,
+            half_views: l.half_views,
+        }
+    };
+    // v1 = first[v0] (one channel).
+    let load = mubuf_idxen(LOAD_FORMAT_X, 1, 0, 8);
+    let half = vop1(V_CVT_F32_F16, 2, vgpr(1));
+    let as_halves = lowered(&[&load[..], &[half]].concat());
+    assert_eq!(as_halves.half_views, [(1, 0)]);
+    let as_float = lowered(&[&load[..], &[vop2(V_ADD_F32, 2, vgpr(1), 1)]].concat());
+    assert!(as_float.half_views.is_empty());
+    // v3 = first[v0] too, read as a whole float: v2 = half(v1) + v3.
+    let mut beside = load.to_vec();
+    beside.extend_from_slice(&mubuf_idxen(LOAD_FORMAT_X, 3, 0, 8));
+    beside.push(half);
+    beside.push(vop2(V_ADD_F32, 2, vgpr(3), 2));
+    assert!(lowered(&beside).half_views.is_empty());
+    // Behind a branch, as the head program has it: where the lanes meet
+    // again the loaded value goes on through a phi, and a later write of
+    // v1 under EXEC reads it only to keep it in the lanes it leaves alone.
+    const SAVE_EXEC: u32 = 0xBE90_037E; // s_mov_b32 s16, exec_lo
+    const ONLY_LANE_0: u32 = 0x7DA4_0080; // v_cmpx_eq_u32 exec_lo, 0, v0
+    const SKIP_3: u32 = 0xBF88_0003; // s_cbranch_execz over the next 3 words
+    const RESTORE_EXEC: u32 = 0xBEFE_0310; // s_mov_b32 exec_lo, s16
+    let zero = vop1(0x01, 2, 128); // v_mov_b32 v2, 0
+    let rewrite = vop1(0x01, 1, 128); // v_mov_b32 v1, 0
+    let branched = lowered(
+        &[
+            &[zero, SAVE_EXEC, ONLY_LANE_0, SKIP_3][..],
+            &load,
+            &[
+                half,
+                RESTORE_EXEC,
+                SAVE_EXEC,
+                ONLY_LANE_0,
+                rewrite,
+                RESTORE_EXEC,
+            ],
+        ]
+        .concat(),
+    );
+    assert_eq!(branched.half_views, [(1, 0)]);
+    if let Ok(mut v) = validator::Validator::load(validator::Validator::default_path().as_deref()) {
+        for shader in [&as_halves, &branched] {
+            let mut dxil = shader.dxil.clone();
+            v.validate(&mut dxil).expect("DXIL validation failed");
+        }
+    }
+    // Element 0 of an R16_UINT buffer holds the half 1.0 (0x3c00).
+    let data = [0x0000_3c00u32, 0];
+    let _one = ONE_AT_A_TIME.lock().unwrap_or_else(|e| e.into_inner());
+    let mut ran = 0;
+    for (label, warp) in [("hardware", false), ("warp", true)] {
+        let Ok(host) = Host::new(warp) else {
+            eprintln!("no SM 6.6 {label} device; skipped");
+            continue;
+        };
+        for (name, shader) in [("straight", &as_halves), ("branched", &branched)] {
+            let (root, pso) = host
+                .compute_pipeline(shader)
+                .unwrap_or_else(|e| panic!("{label}: compute_pipeline failed: {e}"));
+            let mut ud = vec![0u32; shader.user_data_dwords().max(1) as usize];
+            for u in &shader.user_data {
+                ud[u.index as usize] = match (u.root, u.slot) {
+                    (RootType::Table, 1) => 0,
+                    other => panic!("unexpected user data {other:?}"),
+                };
+            }
+            let out = host
+                .dispatch(
+                    &root,
+                    &pso,
+                    &Dispatch {
+                        roots: vec![(0, Root::Constants(ud))],
+                        views: &[
+                            View::Buffer {
+                                data: &data,
+                                stride: Stride::Format(DXGI_FORMAT_R16_UINT, 2),
+                            },
+                            View::RawUav(1),
+                        ],
+                        samplers: &[],
+                        groups: [1, 1, 1],
+                    },
+                )
+                .unwrap_or_else(|e| panic!("{label}: dispatch failed: {e}"));
+            assert_eq!(
+                out[0][0],
+                1.0f32.to_bits(),
+                "{label}, {name}: the half read as an integer's bits"
+            );
+        }
+        ran += 1;
+    }
+    assert!(ran > 0, "no SM 6.6 device available");
+}
