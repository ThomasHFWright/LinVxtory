+// Each lane addresses element v0 of the heap view at s0. Readers export the
+// selected field through a separate raw output UAV in s2. Writers store the
+// lane index, distinguishable from the initial values in every backing dword.
+fn shader(write: bool, field: usize) -> HostShader {
+    let mut code = vec![
+        0x8F04_8500, // s_lshl_b32 s4, s0, 5
+        0xBE86_0301,
+        0xBE87_0380, // s[6:7] = resource heap
+        0xF408_0203,
+        0x0800_0000, // s_load_dwordx4 s[8:11], s[6:7], s4
+    ];
+    let byte = (field * 4) as u32;
+    if write {
+        code.extend([
+            0xBF8C_C07F, // s_waitcnt lgkmcnt(0)
+            0xE070_2000 | byte,
+            0x8002_0000, // store v0 to elem v0, s[8:11]
+        ]);
+    } else {
+        code.extend([
+            0xBE8C_0302,
+            0xBE8D_0380, // s[12:13] = output table
+            0xF408_0406,
+            0xFA00_0000, // s_load_dwordx4 s[16:19], s[12:13], 0
+            0x3402_0082, // v_lshlrev_b32 v1, 2, v0
+            0xBF8C_C07F, // s_waitcnt lgkmcnt(0)
+            0xE030_2000 | byte,
+            0x8002_0200, // load v2 from elem v0, s[8:11]
+            0xBF8C_3F70, // s_waitcnt vmcnt(0)
+            0xE070_1000,
+            0x8004_0201, // store v2 to raw byte v1, s[16:19]
+        ]);
+    }
+    code.push(0xBF81_0000); // s_endpgm
+    let layout = [RootType::Constants, RootType::ResourceHeap, RootType::Table]
+        .into_iter()
+        .enumerate()
+        .map(|(slot, root)| RootParam {
+            root,
+            table_offset_dwords: 0,
+            sgprs: 1,
+            base_register: 0,
+            user_data: slot as u8,
+            slot: slot as u8,
+        })
+        .collect();
+    let output = UsedResource {
+        kind: ResourceType::Uav,
+        root: RootType::Table,
+        table_offset_dwords: 0,
+        lower_bound: 0,
+        stride: 0,
+        dwords: 4,
+        user_data: 2,
+        slot: 2,
+    };
+    let native = NativeProgram {
+        stage: HwStage::Cs,
+        wave64: false,
+        vgprs: 8,
+        layout,
+        used: if write { vec![] } else { vec![output] },
+        io: StageIo::Cs(ComputeIo {
+            rsrc2: 3 << 1,
+            threads: [LANES as u32, 1, 1],
+        }),
+        code: code.iter().flat_map(|w| w.to_le_bytes()).collect(),
+    };
+    let ir = victory_shader::ir::build(&native.code, native.wave64, native.vgprs).unwrap();
+    let st = victory_shader::structure::structure(&ir).unwrap();
+    let res =
+        victory_shader::resource::track(&ir, &native.layout, native.stage, Some(&st)).unwrap();
+    let io = victory_shader::io::raise(&ir, &native).unwrap();
+    let signatures = victory_shader::Container {
+        digest: [0; 16],
+        version: (1, 0),
+        parts: Vec::new(),
+    };
+    let emitted = victory_shader::emit::emit(&signatures, &native, &ir, &st, &io, &res).unwrap();
+    assert_eq!(emitted.root_srvs, [SLOT_STRIDES]);
+    emitted
+}

+struct Kernel {
+    shader: HostShader,
+    root: ID3D12RootSignature,
+    pso: ID3D12PipelineState,
+}
+
+impl Kernel {
+    fn new(device: &victory_shader::emit::host::Host, write: bool, field: usize) -> Self {
+        let shader = shader(write, field);
+        let (root, pso) = device
+            .compute_pipeline(&shader)
+            .expect("overlap compute pipeline");
+        Self { shader, root, pso }
+    }
+
+    fn run(
+        &self,
+        host: &Host,
+        heap: &HostHeap,
+        slot: u32,
+        backing: &std::sync::Arc<crate::shadow::Shadow>,
+        address: u64,
+    ) {
+        let list = host.new_list().unwrap();
+        // As in draw::State::heap_buffers, merge the overlapping view ranges
+        // and defer their refresh until submission.
+        backing.refresh(&list, address, BYTES as u64);
+        let mut ud = vec![0; self.shader.user_data_dwords().max(1) as usize];
+        for u in &self.shader.user_data {
+            ud[u.index as usize] = match u.root {
+                RootType::Constants => slot,
+                RootType::Table => 0,
+                other => panic!("unexpected root {other:?}"),
+            };
+        }
+        // SAFETY: all descriptors, resources and root arguments live through
+        // the waited submission. Each buffer implicitly promotes from COMMON.
+        unsafe {
+            list.list.SetDescriptorHeaps(&[Some(heap.heap.clone())]);
+            list.list.SetComputeRootSignature(&self.root);
+            list.list.SetPipelineState(&self.pso);
+            list.list
+                .SetComputeRoot32BitConstants(0, ud.len() as u32, ud.as_ptr().cast(), 0);
+            list.list
+                .SetComputeRootShaderResourceView(1, heap.strides_gpu().unwrap());
+            list.list.Dispatch(1, 1, 1);
+        }
+        list.close();
+        // Fence completion and implicit buffer decay precede the next run.
+        host.execute(&[&list]);
+        // SAFETY: status query on the live device.
+        unsafe { host.device.GetDeviceRemovedReason().unwrap() };
+    }
+}

+fn mirror(
+    heap: &HostHeap,
+    slot: u32,
+    base: u64,
+    prefix: usize,
+    write: bool,
+    backing: &ID3D12Resource,
+) {
+    let address = base + (prefix * 4) as u64;
+    let mut resource = [0; 14];
+    resource[0] = 1; // D3D12_RESOURCE_DIMENSION_BUFFER
+    resource[4] = (BYTES - prefix * 4) as u32;
+    let mut words = [0; 16];
+    words[10] = address as u32;
+    words[11] = (address >> 32) as u32;
+    let mut record = [0u32; 8];
+    let handle = record.as_mut_ptr() as u64;
+    if write {
+        words[1] = D3D12_UAV_DIMENSION_BUFFER.0 as u32;
+        words[4] = LANES as u32;
+        words[5] = (FIELDS * 4) as u32;
+        crate::descriptor::create_placed_uav(std::ptr::null(), &resource, &words, handle);
+    } else {
+        words[1] = D3D12_SRV_DIMENSION_BUFFER.0 as u32;
+        words[2] = D3D12_DEFAULT_SHADER_4_COMPONENT_MAPPING;
+        words[6] = LANES as u32;
+        words[7] = (FIELDS * 4) as u32;
+        crate::descriptor::create_placed_srv(std::ptr::null(), &resource, &words, handle);
+    }
+    let view = heap.mirror(slot, handle, true).unwrap();
+    let (copy, start, len) = buffer_copy(&view).unwrap();
+    assert_eq!((start, len), (address, (LANES * FIELDS * 4) as u64));
+    assert_eq!(&copy.buffer, backing, "refresh uses a different shadow");
+    assert_eq!(
+        heap.kept.lock().unwrap()[slot as usize].as_ref(),
+        Some(backing),
+        "descriptor uses a different resource"
+    );
+    assert_eq!(
+        metadata(heap, slot),
+        (FIELDS * 4) as u32 | ((prefix as u32) << 28)
+    );

+fn metadata(stride: u32, prefix_words: usize) -> u32 {
+    // Independent encoding of the producer ABI: bits 28..29, in dwords.
+    stride | ((prefix_words as u32) << 28)

+fn picked_buffer_shader(write: bool) -> HostShader {
+    // The heap pick and the idxen load/store are from #2885's overlap_test.
+    let mut words = vec![
+        0x8F04_8500, // s_lshl_b32 s4, s0, 5 (heap slot to byte offset)
+        0xBE86_0301,
+        0xBE87_0380,
+        0xF408_0203,
+        0x0800_0000, // s_load_dwordx4 s[8:11], s[6:7], s4
+    ];
+    if write {
+        words.extend([
+            0x4A04_0081, // v_add_nc_u32 v2, 1, v0
+            0xBF8C_C07F, // wait for the descriptor
+            0xE070_2000,
+            0x8002_0200, // store lane+1 to field 0
+            0xE0C8_2004,
+            0x8002_0200, // atomic add lane+1 to field 1
+            0xE0C8_2004,
+            0x8002_0200, // a second add distinguishes it from a store
+        ]);
+    } else {
+        words.extend([
+            0xBE8C_0302,
+            0xBE8D_0380,
+            0xF408_0406,
+            0xFA00_0000, // s_load_dwordx4 s[16:19], s[12:13], 0
+            0x3402_0082, // v_lshlrev_b32 v1, 2, v0
+            0xBF8C_C07F,
+            0xE030_2004,
+            0x8002_0200, // load field 1 of element v0
+            0xBF8C_3F70,
+            0xE070_1000,
+            0x8004_0201, // store result to output[lane]
+        ]);
+    }
+    words.push(0xBF81_0000);
+    let layout = [RootType::Constants, RootType::ResourceHeap, RootType::Table]
+        .into_iter()
+        .enumerate()
+        .map(|(slot, root)| RootParam {
+            root,
+            table_offset_dwords: 0,
+            sgprs: 1,
+            base_register: 0,
+            user_data: slot as u8,
+            slot: slot as u8,
+        })
+        .collect();
+    let output = UsedResource {
+        kind: ResourceType::Uav,
+        root: RootType::Table,
+        table_offset_dwords: 0,
+        lower_bound: 0,
+        stride: 0,
+        dwords: 4,
+        user_data: 2,
+        slot: 2,
+    };
+    let native = NativeProgram {
+        stage: HwStage::Cs,
+        wave64: false,
+        vgprs: 8,
+        layout,
+        used: if write { vec![] } else { vec![output] },
+        io: StageIo::Cs(ComputeIo {
+            rsrc2: 3 << 1,
+            threads: [LANES as u32, 1, 1],
+        }),
+        code: words.iter().flat_map(|w: &u32| w.to_le_bytes()).collect(),
+    };
+    lowered(&native).0
+}

+fn roots(shader: &HostShader, slot: u32, strides: &[u32]) -> Vec<(u32, Root)> {
+    assert_eq!(shader.root_srvs, [SLOT_STRIDES]);
+    assert!(shader.root_cbvs.is_empty() && shader.root_uavs.is_empty());
+    let mut ud = vec![0; shader.user_data_dwords().max(1) as usize];
+    for u in &shader.user_data {
+        ud[u.index as usize] = match u.root {
+            RootType::Constants => slot,
+            RootType::Table => 0,
+            other => panic!("unexpected user data {other:?}"),
+        };
+    }
+    vec![
+        (0, Root::Constants(ud)),
+        (
+            1,
+            Root::View {
+                srv: true,
+                data: strides.to_vec(),
+            },
+        ),
+    ]

+#[test]
+fn heap_reads_use_the_selected_views_stride_and_prefix() {
+    let shader = picked_buffer_shader(false);
+    let mut strides = vec![4]; // output UAV at slot 0
+    let buffers: Vec<Vec<u32>> = STRIDES
+        .iter()
+        .enumerate()
+        .map(|(prefix, &stride)| {
+            strides.push(metadata(stride, prefix));
+            let n = prefix + LANES * stride as usize / 4;
+            (0..n)
+                .map(|i| 10000 * (prefix as u32 + 1) + i as u32)
+                .collect()
+        })
+        .collect();
+    let mut views = vec![View::RawUav(LANES as u32)];
+    views.extend(buffers.iter().map(|data| View::Buffer {
+        data,
+        stride: Stride::Raw,
+    }));
+    on_adapters(|host, label| {
+        let (root, pso) = host.compute_pipeline(&shader).expect("compute pipeline");
+        for (prefix, &stride) in STRIDES.iter().enumerate() {
+            let words = host
+                .dispatch(
+                    &root,
+                    &pso,
+                    &Dispatch {
+                        roots: roots(&shader, prefix as u32 + 1, &strides),
+                        views: &views,
+                        samplers: &[],
+                        groups: [1, 1, 1],
+                    },
+                )
+                .expect("readback");

+        }
+    });
