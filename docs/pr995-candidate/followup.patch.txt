+/// A clean run can cross reservation boundaries, where VirtualProtect fails
+/// atomically. Retry each block so every accessible block is opened.
+fn reopen_clean_run(first: usize, end: usize, mut protect: impl FnMut(usize, usize) -> bool) {
+    let count = end - first;
+    if !protect(first * BLOCK, count * BLOCK) && count > 1 {
+        for block in first..end {
+            protect(block * BLOCK, BLOCK);
+        }
+    }
+}
+
+#[cfg(test)]
+mod reopen_tests {
+    use super::{BLOCK, reopen_clean_run};
+
+    #[test]
+    fn clean_run_fallback_matches_per_block_protection() {
+        for mask in 0u16..256 {
+            for first in 0..8 {
+                for end in first + 1..=8 {
+                    let mut opened = 0;
+                    let mut calls = 0;
+                    reopen_clean_run(first, end, |at, bytes| {
+                        calls += 1;
+                        let bits = ((1u16 << (bytes / BLOCK)) - 1) << (at / BLOCK);
+                        if bits & mask != bits {
+                            return false;
+                        }
+                        opened |= bits;
+                        true
+                    });
+                    let requested = ((1u16 << (end - first)) - 1) << first;
+                    assert_eq!(opened, mask & requested);
+                    let fallback = requested & mask != requested && end - first > 1;
+                    assert_eq!(calls, if fallback { 1 + end - first } else { 1 });
+                }
+            }
+        }
+    }
+}
